\begin{algorithm}[t]
    \caption{Variance-Aware MeanFlow Training}
    \label{alg: va-mf}
    \begin{algorithmic}
        \Require Dataset $\gD$, EMA decay $\mu$, anchor weight $\lambda$, anchor interval $[\delta_{\min},\delta_{\max}]$
        \Require Schedule $\sigma_{t}$, number of probes $K$
        \State Initialize $\bar{\vtheta}\gets\vtheta$
        \For{$k=1,2,\ldots$}
            \State Sample $(\vx_{0},\vx_{1})\sim\gD\times\gN(\bm{0},\mI_{d})$,\; $t\sim\gU[0,1]$,\; $r\sim\gU[0,t]$,\; $\delta\sim\gU[\delta_{\min},\delta_{\max}]$
            \State $\vx_{t}\gets(1-t)\vx_{0}+t\,\vx_{1}$,\quad $\vv_{\text{cond}}\gets\vx_{1}-\vx_{0}$
            \State $\vv_{\text{tang}}\gets\texttt{sg}[\vu_{\bar{\vtheta}}(\vx_{t},t,t)]$ \Comment{EMA tangent}
            \State $V_{\vtheta}\gets\vu_{\vtheta}(\vx_{t},r,t)+(t{-}r)\,\texttt{sg}[\texttt{JVP}(\vu_{\vtheta},(\vx_{t},r,t),(\vv_{\text{tang}},0,1))]$
            \State Sample $K$ probes $\{\vz_{i}\}_{i=1}^{K}\!\sim\!\gN(\bm{0},\mI_{d})$
            \State $\widehat{\Tr(\mJ\mJ^{\!\top})}\gets\frac{1}{K}\sum_{i=1}^{K}\bigl\|(t{-}r)\,\partial_{\vz}\vu_{\vtheta}(\vx_{t},r,t)\,\vz_{i}-\vz_{i}\bigr\|_{2}^{2}$
            \State $w\gets 1\,/\,\bigl(1+\sigma_{t}^{2}\,\texttt{sg}[\widehat{\Tr(\mJ\mJ^{\!\top})}]/d\bigr)$ \Comment{trace weight}
            \State $\ell_{\text{MF}}\gets w\cdot\|V_{\vtheta}-\vv_{\text{cond}}\|_{2}^{2}$
            \State $\ell_{\text{FM}}\gets\|\vu_{\vtheta}(\vx_{t},t{-}\delta,t)-\vv_{\text{cond}}\|_{2}^{2}$ \Comment{FM anchor}
            \State $\vtheta\gets\vtheta-\eta\,\nabla_{\vtheta}(\ell_{\text{MF}}+\lambda\,\ell_{\text{FM}})$
            \State $\bar{\vtheta}\gets\mu\bar{\vtheta}+(1-\mu)\vtheta$ \Comment{EMA update}
        \EndFor
        \State \Return $\vtheta$ \Comment{Generate: $\hat{\vx}_{0}=\vx_{1}-\vu_{\vtheta}(\vx_{1},0,1)$}
    \end{algorithmic}
\end{algorithm}
