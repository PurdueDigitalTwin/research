    \section{Proof of Lemma}
    \label[appendix]{appx: first}
    %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

    \begin{lemma}
        Given vector fields $\vv(\vx,t\mid{\vx_{0}})$ that generate probability paths $p(\vx,t\mid{\vx_{0}})$, for any data distribution $\rvx_{0}\sim p(\rvx_{0})$, the marginal velocity field is equal to the expected conditional velocity field
        \begin{equation*}
            \vv(\vx,t)=\E_{\vx_{0}\sim{p(\vx_0,t\mid{\vx})}}
            \left[\vv(\vx,t\mid{\vx_{0}})\right].
        \end{equation*}
    \end{lemma}

    \begin{proof}
        A sufficient and necessary condition for a vector field to generate a probability path is the \textit{continuity equation}~\cite{villani2009optimaltransport}, which is a partial differential equation (PDE) that writes
        \begin{equation}
            \frac{\partial}{\partial t}p(\vx,t)+
            \text{div}\left(p(\vx,t)\vv(\vx,t)\right)=0,
        \end{equation}
        where $\text{div}(f)$ denotes the divergence of function $f$. The marginal velocity field generates the marginal density $p(\vx,t)$ via the continuity equation. By the law of total probability and the definition of the conditional velocity field:
        \begin{equation}
            p(\vx,t)\,\vv(\vx,t)=\int p(\vx,t\mid\vx_{0})\,\vv(\vx,t\mid\vx_{0})\,p(\vx_{0})\,\mathrm{d}\vx_{0},
        \end{equation}
        since both sides satisfy the continuity equation and generate the same marginal path $p(\vx,t)$. Dividing by $p(\vx,t)=\int p(\vx,t\mid\vx_{0})\,p(\vx_{0})\,\mathrm{d}\vx_{0}$ and applying Bayes' rule $p(\vx_{0}\mid\vx_{t}=\vx)=p(\vx,t\mid\vx_{0})\,p(\vx_{0})/p(\vx,t)$:
        \begin{equation}
            \vv(\vx,t)=\int\vv(\vx,t\mid\vx_{0})\,p(\vx_{0}\mid\vx_{t}=\vx)\,\mathrm{d}\vx_{0}=\E_{\rvx_{0}\mid\rvx_{t}=\vx}\!\left[\vv(\vx,t\mid\rvx_{0})\right].
        \end{equation}
    \end{proof}

    \subsection{Proof of~\Cref{thm: jacobi-variance} (Jacobian Variance Amplification)}

    \begin{theorem}[Jacobian Variance Amplification]
        The Jacobi factor amplifies the gradient variance of the original MeanFlow loss:
        \begin{equation*}
            \Var\!\left[\nabla_{\vtheta}\ell_{\text{MF}}(\vtheta)\mid\vx_{t}\right]\;\propto\;\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right).
        \end{equation*}
    \end{theorem}

    \begin{proof}
        Under the stop-gradient operator, the compound prediction $V_{\vtheta}^{\texttt{sg}}$ depends on $\vtheta$ only through $\vu_{\vtheta}(\vx_{t},r,t)$. Let $\nabla_{\vtheta}\vu_{\vtheta}\in\mathbb{R}^{d\times p}$ denote the parameter Jacobian of the network output. The per-step gradient is
        \begin{equation}
            \nabla_{\vtheta}\ell_{\text{MF}}=2\left(\nabla_{\vtheta}\vu_{\vtheta}\right)^{\!\top}\!\left(\vr_{\vtheta}^{\texttt{sg}}+\mJ\vv'\right),
            \label{eq: per-step-grad}
        \end{equation}
        where $\vr_{\vtheta}^{\texttt{sg}}$ is the mean-field residual and $\mJ=(t-r)\partial_{\vz}\vu_{\vtheta}-\mI_{d}$ is the Jacobi factor. All three quantities---$\nabla_{\vtheta}\vu_{\vtheta}$, $\vr_{\vtheta}^{\texttt{sg}}$, and $\mJ$---are deterministic conditioned on $\vx_{t}$.

        Since $\E_{\vx_{0}\mid\vx_{t}}[\vv']=\bm{0}$ by~\eqref{eq: margv-condv-relation}, the conditional mean gradient is:
        \begin{equation}
            \E\!\left[\nabla_{\vtheta}\ell\mid\vx_{t}\right]=2\left(\nabla_{\vtheta}\vu_{\vtheta}\right)^{\!\top}\vr_{\vtheta}^{\texttt{sg}}.
        \end{equation}
        The centered gradient is $\nabla_{\vtheta}\ell-\E[\nabla_{\vtheta}\ell\mid\vx_{t}]=2(\nabla_{\vtheta}\vu_{\vtheta})^{\!\top}\mJ\vv'$, so the conditional variance is:
        \begin{align}
            \Var\!\left[\nabla_{\vtheta}\ell\mid\vx_{t}\right]
            &=\E\!\left[\left\|2(\nabla_{\vtheta}\vu_{\vtheta})^{\!\top}\mJ\vv'\right\|^{2}\mid\vx_{t}\right] \nonumber\\
            &=4\,\E\!\left[(\vv')^{\!\top}\mJ^{\!\top}\underbrace{(\nabla_{\vtheta}\vu_{\vtheta})(\nabla_{\vtheta}\vu_{\vtheta})^{\!\top}}_{\mG_{\vtheta}\;\succeq\;\bm{0}}\mJ\vv'\;\middle|\;\vx_{t}\right] \nonumber\\
            &=4\,\Tr\!\left(\mG_{\vtheta}\,\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right)\;\propto\;\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right),
        \end{align}
        where $\mG_{\vtheta}=(\nabla_{\vtheta}\vu_{\vtheta})(\nabla_{\vtheta}\vu_{\vtheta})^{\!\top}\in\mathbb{R}^{d\times d}$ is the parameter-space Gram matrix and the last step uses $\mG_{\vtheta}\succeq\bm{0}$ to establish proportionality via the cyclic property of the trace.
    \end{proof}

    \subsection{Proof of~\Cref{thm: semi-gradient-gap} (Semi-Gradient Gap)}

    \begin{theorem}[Semi-Gradient Gap]
        The gradients of the MeanFlow loss with and without the stop-gradient operator differ by
        \begin{equation*}
            \underbrace{2(t{-}r)\,\E_{r,t,\vx_{0},\vx_{1}}\!\left[\nabla_{\vtheta}\!\left(\partial_{\vx_{t}}\vu_{\vtheta}\!\cdot\!\vv(\vx_{t},t)+\partial_{t}\vu_{\vtheta}\right)\!\cdot\!\vr_{\vtheta}\right]}_{\text{mean-field gradient difference}}
            +\underbrace{\E_{r,t,\vx_{0},\vx_{1}}\!\left[\nabla_{\vtheta}\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right)\right]}_{\text{variance-driven gradient difference}}.
        \end{equation*}
    \end{theorem}

    \begin{proof}
        The MeanFlow loss conditioned on $\vx_{t}$ decomposes as (cf.~\eqref{eq: per-sample-expand}):
        \begin{equation}
            \E_{\vx_{0}\mid\vx_{t}}\!\left[\ell_{\text{MF}}\right]=\left\|\vr_{\vtheta}^{\texttt{sg}}\right\|^{2}+\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right).
            \label{eq: cond-loss-decomp}
        \end{equation}
        \textbf{Without stop-gradient.} Both terms depend on $\vtheta$ (through $\vu_{\vtheta}$ and $\partial_{\vz}\vu_{\vtheta}$), so the full gradient is:
        \begin{equation}
            \nabla_{\vtheta}\E\!\left[\ell\right]=\nabla_{\vtheta}\left\|\vr_{\vtheta}\right\|^{2}+\nabla_{\vtheta}\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right).
        \end{equation}
        \textbf{With stop-gradient.} The JVP terms in $\vr_{\vtheta}^{\texttt{sg}}$ and $\mJ$ are treated as constants, so:
        \begin{equation}
            \nabla_{\vtheta}^{\texttt{sg}}\left\|\vr^{\texttt{sg}}\right\|^{2}=2\left(\nabla_{\vtheta}\vu_{\vtheta}\right)^{\!\top}\vr^{\texttt{sg}},\qquad
            \nabla_{\vtheta}^{\texttt{sg}}\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right)=\bm{0}.
        \end{equation}
        The gap between the full and stop-gradient gradients is therefore:
        \begin{align}
            &\nabla_{\vtheta}\E\!\left[\ell\right]-\nabla_{\vtheta}^{\texttt{sg}}\E\!\left[\ell\right] \nonumber\\
            &\quad=\underbrace{\nabla_{\vtheta}\left\|\vr_{\vtheta}\right\|^{2}-2\left(\nabla_{\vtheta}\vu_{\vtheta}\right)^{\!\top}\vr^{\texttt{sg}}}_{\text{mean-field gradient difference}}+\underbrace{\nabla_{\vtheta}\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right)}_{\text{variance-driven difference}}.
        \end{align}
        Expanding the first term: $\nabla_{\vtheta}\|\vr\|^{2}=2(\nabla_{\vtheta}\vr_{\vtheta})^{\!\top}\vr_{\vtheta}$, and $\nabla_{\vtheta}\vr_{\vtheta}$ includes the derivative of the JVP terms $(t-r)(\partial_{\vz}\vu_{\vtheta}\cdot\vv+\partial_{t}\vu_{\vtheta})$ w.r.t.\ $\vtheta$, yielding the second-order mean-field difference $2(t-r)\E[\nabla_{\vtheta}(\partial_{\vz}\vu_{\vtheta}\cdot\vv+\partial_{t}\vu_{\vtheta})\cdot\vr_{\vtheta}]$. At convergence $\vr_{\vtheta}\to\bm{0}$, this term vanishes, and the gap is dominated by the variance-driven term $\nabla_{\vtheta}\Tr(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top})$.
    \end{proof}

    \subsection{Proof of~\Cref{thm: material-derivative-expansion} (Flow Acceleration Expansion)}

    \begin{theorem}[Flow Acceleration Expansion of the Curvature Gap]
        Let $\vv(\vx,t)\!\in\!C^{2}$. The curvature gap $\Delta(\vx_{t},r,t)\!:=\!\vu(\vx_{t},r,t)-\vv(\vx_{t},t)$ admits a Taylor expansion at $(\vx_{t},t)$ along the flow:
        \begin{equation*}
            \Delta(\vx_{t},r,t)\;=\;-\frac{(t{-}r)}{2}\frac{\mathrm{D}\vv}{\mathrm{D}t}(\vx_{t},t)+\gR_{2}(\vx_{t},r,t),
        \end{equation*}
        where $\frac{\mathrm{D}\vv}{\mathrm{D}t}\!\triangleq\!\partial_{t}\vv+(\nabla_{\vx}\vv)\,\vv$ is the material derivative (\textit{a.k.a.\ flow acceleration}) and $\gR_{2}\!=\!\gO((t{-}r)^{2})$.
        \label[theorem]{thm: material-derivative-expansion}
    \end{theorem}

    \begin{proof}
        Let $\vx_{\tau}$ solve the ODE $\frac{\mathrm{d}\vx_{\tau}}{\mathrm{d}\tau}=\vv(\vx_{\tau},\tau)$ with terminal condition $\vx_{t}=\vx_{t}$. The curvature gap $\Delta=\vu(\vx_{t},r,t)-\vv(\vx_{t},t)$ can be written as:
        \begin{equation}
            \Delta(\vx_{t},r,t)=\frac{1}{t-r}\int_{r}^{t}\!\vv(\vx_{\tau},\tau)\,\mathrm{d}\tau-\vv(\vx_{t},t)=\frac{1}{t-r}\int_{r}^{t}\!\left[\vv(\vx_{\tau},\tau)-\vv(\vx_{t},t)\right]\mathrm{d}\tau.
        \end{equation}
        Define $h=\tau-t\in[-(t-r),0]$. Since $\vv\in C^{2}$, we Taylor-expand $\vv(\vx_{\tau},\tau)$ about $(\vx_{t},t)$ along the ODE trajectory:
        \begin{equation}
            \vv(\vx_{\tau},\tau)=\vv(\vx_{t},t)+\frac{\mathrm{D}\vv}{\mathrm{D}t}(\vx_{t},t)\,h+\frac{1}{2}\frac{\mathrm{D}^{2}\vv}{\mathrm{D}t^{2}}(\vx_{t},t)\,h^{2}+O(h^{3}),
        \end{equation}
        where $\frac{\mathrm{D}}{\mathrm{D}t}$ denotes the material derivative along the flow. Substituting and integrating over $h\in[-(t-r),0]$:
        \begin{align}
            \Delta&=\frac{1}{t-r}\int_{-(t-r)}^{0}\!\left[\frac{\mathrm{D}\vv}{\mathrm{D}t}\,h+\frac{1}{2}\frac{\mathrm{D}^{2}\vv}{\mathrm{D}t^{2}}\,h^{2}+O(h^{3})\right]\mathrm{d}h \nonumber\\
            &=\frac{1}{t-r}\left[\frac{\mathrm{D}\vv}{\mathrm{D}t}\cdot\frac{h^{2}}{2}\bigg|_{-(t-r)}^{0}+\frac{1}{2}\frac{\mathrm{D}^{2}\vv}{\mathrm{D}t^{2}}\cdot\frac{h^{3}}{3}\bigg|_{-(t-r)}^{0}+O\!\left((t-r)^{4}\right)\right] \nonumber\\
            &=\frac{1}{t-r}\left[-\frac{(t-r)^{2}}{2}\frac{\mathrm{D}\vv}{\mathrm{D}t}+\frac{(t-r)^{3}}{6}\frac{\mathrm{D}^{2}\vv}{\mathrm{D}t^{2}}+O\!\left((t-r)^{4}\right)\right] \nonumber\\
            &=-\frac{(t-r)}{2}\frac{\mathrm{D}\vv}{\mathrm{D}t}(\vx_{t},t)+\frac{(t-r)^{2}}{6}\frac{\mathrm{D}^{2}\vv}{\mathrm{D}t^{2}}(\vx_{t},t)+\gR_{3}(\vx_{t},r,t),
        \end{align}
        where $\gR_{3}=O((t-r)^{3})$ is the remainder.
    \end{proof}

    \subsection{Proof of~\Cref{prop: vamf-grad} (Gradient of VaMF Loss)}

    \begin{proposition}[Gradient of VaMF Loss]
        The gradient of $\gL_{\text{MF}}^{\text{EMA}}$ takes the form
        \begin{equation*}
            \nabla_{\vtheta}\gL_{\text{MF}}^{\text{EMA}}\;=\;2\,\E\!\left[\nabla_{\vtheta}\vu_{\vtheta}(\vx_{t},r,t)\cdot\tilde{\vr}^{\text{EMA}}\right],
        \end{equation*}
        where $\tilde{\vr}^{\text{EMA}}\!:=\!V_{\vtheta}^{\text{EMA}}\!-\!\vv(\vx_{t},t)$ is the mean-field residual under the EMA tangent, and no $\mJ\vv'$ noise term appears.
    \end{proposition}

    \begin{proof}
        The EMA velocity anchor $\vu_{\bar{\vtheta}}(\vx_{t},t,t)$ is deterministic given $\vx_{t}$, since the EMA parameters $\bar{\vtheta}$ are fixed during the gradient step. Under stop-gradient, $V_{\vtheta}^{\text{EMA}}$ depends on $\vtheta$ only through $\vu_{\vtheta}(\vx_{t},r,t)$. By the same argument as~\eqref{eq: per-step-grad}, the per-step gradient is:
        \begin{equation}
            \nabla_{\vtheta}\ell=2\left(\nabla_{\vtheta}\vu_{\vtheta}\right)^{\!\top}\!\left(V_{\vtheta}^{\text{EMA}}-\vv_{\text{cond}}\right).
        \end{equation}
        Writing $\vv_{\text{cond}}=\vv(\vx_{t},t)+\vv'$ and noting that $\tilde{\vr}^{\text{EMA}}:=V_{\vtheta}^{\text{EMA}}-\vv(\vx_{t},t)$ is deterministic given $\vx_{t}$:
        \begin{equation}
            \E_{\vx_{0}\mid\vx_{t}}\!\left[\nabla_{\vtheta}\ell\right]=2\left(\nabla_{\vtheta}\vu_{\vtheta}\right)^{\!\top}\tilde{\vr}^{\text{EMA}},
        \end{equation}
        since $\E[\vv'\mid\vx_{t}]=\bm{0}$. The tower property $\nabla_{\vtheta}\gL_{\text{MF}}^{\text{EMA}}=\E_{r,t,\vx_{t}}\!\left[\E_{\vx_{0}\mid\vx_{t}}[\nabla_{\vtheta}\ell]\right]$ gives the stated result. Crucially, no $\mJ\vv'$ term appears because the EMA tangent is deterministic---compare with the original MeanFlow gradient~\eqref{eq: per-step-grad} where the stochastic tangent produces the $\mJ\vv'$ noise.
    \end{proof}

    \subsection{Proof of~\Cref{prop: bias-var} (Bias-Variance Tradeoff)}

    \begin{proposition}[Bias-Variance Tradeoff]
        The EMA mean-field residual decomposes as
        \begin{equation*}
            \tilde{\vr}^{\text{EMA}}\;=\;\vr_{\vtheta}\;+\;(t{-}r)\,\partial_{\vz}\vu_{\vtheta}\!\left(\vu_{\bar{\vtheta}}(\vx_{t},t,t)-\vv(\vx_{t},t)\right),
        \end{equation*}
        where $\vr_{\vtheta}$ is the true MeanFlow residual from~\eqref{eq: per-sample-expand}. At convergence $\vr_{\vtheta}\!\to\!\bm{0}$ and the FM anchor drives $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\!\to\!\vv(\vx_{t},t)$, giving $\tilde{\vr}^{\text{EMA}}\!\to\!\bm{0}$.
    \end{proposition}

    \begin{proof}
        Expanding $V_{\vtheta}^{\text{EMA}}$ from~\eqref{eq: ema-mf-loss}, the value of the compound prediction (ignoring the stop-gradient, which does not affect values) is:
        \begin{equation}
            V_{\vtheta}^{\text{EMA}}=\vu_{\vtheta}+(t-r)\!\left(\partial_{\vz}\vu_{\vtheta}\cdot\vu_{\bar{\vtheta}}(\vx_{t},t,t)+\partial_{t}\vu_{\vtheta}\right).
        \end{equation}
        The true MeanFlow residual (with the marginal velocity as tangent) is:
        \begin{equation}
            \vr_{\vtheta}=\vu_{\vtheta}+(t-r)\!\left(\partial_{\vz}\vu_{\vtheta}\cdot\vv(\vx_{t},t)+\partial_{t}\vu_{\vtheta}\right)-\vv(\vx_{t},t).
        \end{equation}
        Subtracting:
        \begin{align}
            \tilde{\vr}^{\text{EMA}}&=V_{\vtheta}^{\text{EMA}}-\vv(\vx_{t},t) \nonumber\\
            &=\vr_{\vtheta}+(t-r)\,\partial_{\vz}\vu_{\vtheta}\!\left(\vu_{\bar{\vtheta}}(\vx_{t},t,t)-\vv(\vx_{t},t)\right).
        \end{align}
        At convergence, the true residual $\vr_{\vtheta}\to\bm{0}$ (the MeanFlow identity is satisfied) and the FM anchor loss~\eqref{eq: fm-anchor} supervises the boundary condition, driving $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\to\vv(\vx_{t},t)$ and hence $\tilde{\vr}^{\text{EMA}}\to\bm{0}$.
    \end{proof}

    \subsection{Proof of~\Cref{prop: target-tangent-asymmetry} (Target--Tangent Bias Asymmetry)}

    \begin{proposition}[Target--Tangent Bias Asymmetry]
        Let $\hat{\vv}\!=\!\vu_{\bar{\vtheta}}(\vx_{t},t,t)$ be a deterministic proxy with bias $\vb\!\triangleq\!\hat{\vv}-\vv(\vx_{t},t)$. For the tangent-replacement loss (VaMF), the loss minimum $\vu^{\star}$ satisfies $\vu^{\star}(\vx_{t},r,t)-\vv(\vx_{t},t)\!=\!-(t{-}r)\,\partial_{\vz}\vu^{\star}\,\vb+\gO(\|\vb\|^{2})$. For the target-replacement loss, the loss minimum at $r\!=\!t$ satisfies $\vu^{\star}(\vx_{t},t,t)-\vv(\vx_{t},t)\!=\!\vb$.
    \end{proposition}

    \begin{proof}
        \textbf{Tangent replacement.} Substituting $\hat{\vv}$ for $\vv_{\text{cond}}$ only in the JVP tangent (target unchanged) gives the per-sample compound prediction
        \begin{equation*}
            V_{\vtheta}^{\text{EMA}}=\vu_{\vtheta}+(t-r)\!\left(\partial_{\vz}\vu_{\vtheta}\cdot\hat{\vv}+\partial_{t}\vu_{\vtheta}\right),
        \end{equation*}
        and the loss is $\E\|V_{\vtheta}^{\text{EMA}}-\vv_{\text{cond}}\|^{2}$. By stationarity, $\E[V_{\vtheta}^{\text{EMA}}\!-\!\vv_{\text{cond}}\mid\vx_{t}]\!=\!\bm{0}$ at the minimum. Taking conditional expectation and using $\E[\vv_{\text{cond}}\mid\vx_{t}]\!=\!\vv(\vx_{t},t)$,
        \begin{equation*}
            \vu^{\star}+(t-r)\!\left(\partial_{\vz}\vu^{\star}\cdot\hat{\vv}+\partial_{t}\vu^{\star}\right)=\vv(\vx_{t},t).
        \end{equation*}
        At the same point, the \emph{true} MeanFlow identity (with the marginal velocity as tangent) reads
        \begin{equation*}
            \vu^{\diamond}+(t-r)\!\left(\partial_{\vz}\vu^{\diamond}\cdot\vv+\partial_{t}\vu^{\diamond}\right)=\vv(\vx_{t},t),
        \end{equation*}
        with $\vu^{\diamond}(\vx_{t},t,t)=\vv(\vx_{t},t)$. Subtracting and writing $\vu^{\star}=\vu^{\diamond}+\Delta$,
        \begin{equation*}
            \Delta+(t-r)\,\partial_{\vz}\vu^{\star}\,(\hat{\vv}-\vv)+(t-r)\,\partial_{t}\Delta+(t-r)\,(\partial_{\vz}\Delta)\,\vv=\bm{0}.
        \end{equation*}
        Evaluating at $r\!=\!t$ gives $\Delta(\vx_{t},t,t)\!=\!\bm{0}$ (boundary condition recovered exactly). For $r\!<\!t$, the leading term in $\vb\!=\!\hat{\vv}-\vv$ gives
        \begin{equation*}
            \Delta(\vx_{t},r,t)=-(t-r)\,\partial_{\vz}\vu^{\star}\,\vb+\gO(\|\vb\|^{2}),
        \end{equation*}
        i.e., the stationary error is $\gO((t-r)\,\|\partial_{\vz}\vu\|\,\|\vb\|)$ and vanishes at the boundary $r\!\to\!t$.
        \\[2pt]
        \textbf{Target replacement.} Substituting $\hat{\vv}$ for $\vv_{\text{cond}}$ in the regression target yields the loss $\E\|\vu_{\vtheta}+(t-r)\,\mathrm{D}\vu/\mathrm{D}t-\hat{\vv}\|^{2}$. By stationarity, the loss minimum satisfies $\E[\vu^{\star}+(t-r)\,\mathrm{D}\vu^{\star}/\mathrm{D}t-\hat{\vv}\mid\vx_{t}]\!=\!\bm{0}$. Specializing to $r\!=\!t$, the gap term vanishes and the equation collapses to $\vu^{\star}(\vx_{t},t,t)\!=\!\hat{\vv}\!=\!\vv(\vx_{t},t)+\vb$, so the stationary error at the boundary is exactly $\vb$, independent of $(t,r)$.
        \\[2pt]
        \textbf{Asymmetry.} Tangent replacement carries a multiplicative $(t-r)$ factor on the bias and is therefore pinned to zero at the boundary $r\!=\!t$ that the FM anchor enforces; target replacement carries no such factor and the bias persists at every $t$, requiring a boundary regularizer at \emph{every} $(r,t)$ to remove.
    \end{proof}

    \subsection{Proof of~\Cref{thm: control-variate-optimum} (Optimal Control-Variate Coefficient)}

    \begin{theorem}[Optimal control-variate coefficient]
        Under the scalar-isotropic approximation $\Sigma_{\vv'}\!\approx\!\sigma^{2}\mI_{d}$ and $\mJ\!\approx\!\kappa\mI_{d}$, the conditional MSE of $\boldsymbol{g}^{(\beta)}$ in~\eqref{eq: g-beta} is
        \begin{equation*}
            M(\beta) \;=\; \beta^{2}(\kappa{+}1)^{2}\|\vb\|^{2} \;+\; \sigma^{2}d\bigl((1{-}\beta)\kappa - \beta\bigr)^{2},
        \end{equation*}
        with unique minimizer $\beta^{\star}\!=\!\frac{\kappa}{\kappa+1}\!\cdot\!\frac{\sigma^{2}d}{\sigma^{2}d+\|\vb\|^{2}}$ and optimum value $M(\beta^{\star})\!=\!\sigma^{2}d\,\kappa^{2}\,\|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2})$. For all $\kappa,\|\vb\|^{2}\!>\!0$, the optimum strictly dominates both $M(0)\!=\!\sigma^{2}\kappa^{2}d$ and $M(1)\!=\!(\kappa{+}1)^{2}\|\vb\|^{2}+\sigma^{2}d$.
    \end{theorem}

    \begin{proof}
        \textbf{Step 1: bias-variance decomposition.} From~\eqref{eq: g-beta},
        \begin{equation*}
            \boldsymbol{g}^{(\beta)} \;=\; 2\,\boldsymbol{g}^{\!\top}\!\Bigl[\bar{\vr} \;+\; \bigl((1{-}\beta)\,\mJ - \beta\,\mI\bigr)\,\vv' \;+\; \beta\,(\mJ{+}\mI)\,\vb\Bigr].
        \end{equation*}
        Using $\E_{\vx_{0}\mid\vx_{t}}[\vv']\!=\!\bm{0}$, the conditional mean is $\E[\boldsymbol{g}^{(\beta)}\mid\vx_{t}]\!=\!2\boldsymbol{g}^{\!\top}(\bar{\vr}+\beta(\mJ{+}\mI)\vb)$ and the centered noise is $-2\boldsymbol{g}^{\!\top}\bigl((1{-}\beta)\mJ-\beta\mI\bigr)\vv'$ (relative to the ideal target $2\boldsymbol{g}^{\!\top}\bar{\vr}$). The conditional MSE w.r.t.\ the ideal target therefore decomposes as
        \begin{equation*}
            \mathrm{MSE} \;=\; \underbrace{4\beta^{2}\,\bigl\|\boldsymbol{g}^{\!\top}(\mJ{+}\mI)\vb\bigr\|^{2}}_{\text{bias}^{2}} \;+\; \underbrace{4\,\Tr\!\Bigl(\boldsymbol{g}\boldsymbol{g}^{\!\top}\bigl((1{-}\beta)\mJ-\beta\mI\bigr)\Sigma_{\vv'}\bigl((1{-}\beta)\mJ-\beta\mI\bigr)^{\!\top}\Bigr)}_{\text{variance}}.
        \end{equation*}
        \textbf{Step 2: scalar approximation.} Substituting $\mJ\!=\!\kappa\mI_{d}$ and $\Sigma_{\vv'}\!=\!\sigma^{2}\mI_{d}$, and absorbing the parameter-direction normalization $\Tr(\boldsymbol{g}\boldsymbol{g}^{\!\top})\!=\!\|\boldsymbol{g}\|^{2}_{F}$ into a per-component MSE,
        \begin{equation*}
            M(\beta) \;\propto\; \beta^{2}(\kappa{+}1)^{2}\|\vb\|^{2} \;+\; \sigma^{2}d\bigl((1{-}\beta)\kappa - \beta\bigr)^{2}.
        \end{equation*}
        \textbf{Step 3: minimization.} Differentiating and setting to zero,
        \begin{align*}
            \frac{\partial M}{\partial \beta} \;=\;& 2\beta(\kappa{+}1)^{2}\|\vb\|^{2} - 2(\kappa{+}1)\sigma^{2}d\bigl((1{-}\beta)\kappa-\beta\bigr) \;=\; 0\\
            \Longleftrightarrow\quad& \beta(\kappa{+}1)\bigl(\sigma^{2}d+\|\vb\|^{2}\bigr) \;=\; \sigma^{2}d\,\kappa.
        \end{align*}
        Solving yields $\beta^{\star}\!=\!\sigma^{2}d\,\kappa/[(\kappa{+}1)(\sigma^{2}d+\|\vb\|^{2})]$, which factors as the noise-cancellation $\kappa/(\kappa{+}1)$ times the James-Stein shrinkage $\sigma^{2}d/(\sigma^{2}d+\|\vb\|^{2})$.

        \textbf{Step 4: optimum value.} At $\beta^{\star}$, $(1{-}\beta^{\star})\kappa - \beta^{\star} = \kappa\,\|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2})$. Substituting back,
        \begin{equation*}
            M(\beta^{\star}) \;=\; \frac{\sigma^{4}d^{2}\kappa^{2}\|\vb\|^{2}+\sigma^{2}d\,\kappa^{2}\|\vb\|^{4}}{(\sigma^{2}d+\|\vb\|^{2})^{2}} \;=\; \frac{\sigma^{2}d\,\kappa^{2}\,\|\vb\|^{2}}{\sigma^{2}d+\|\vb\|^{2}}.
        \end{equation*}
        \textbf{Step 5: strict dominance over corners.} For $\kappa,\|\vb\|^{2}\!>\!0$:
        $M(\beta^{\star})/M(0) \!=\! \|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2}) \!<\! 1$, so $M(\beta^{\star})\!<\!M(0)$.
        For the $M(1)$ comparison, $M(1)\!-\!M(\beta^{\star})\!=\!(\kappa{+}1)^{2}\|\vb\|^{2}+\sigma^{2}d-\sigma^{2}d\kappa^{2}\|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2})$. The first two terms exceed $\sigma^{2}d\kappa^{2}\|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2})$ since $(\kappa{+}1)^{2}\!>\!\kappa^{2}$ for $\kappa\!>\!0$.
    \end{proof}

    \subsection{Derivation of the Trace Weight}
    \label{appx: blue-scalar}

    The control-variate optimum of~\Cref{thm: control-variate-optimum} suggests an alternative \textit{loss-side} reweighting: rather than tuning the tangent coefficient $\beta$, equalize per-sample loss magnitudes by downweighting high-variance samples directly. We formalize this as a \textit{trace weight}
    \begin{equation}
        w(\vx_{t},r,t)\;:=\;\frac{1}{1+\sigma_{t}^{2}\,\texttt{sg}\!\left[\Tr(\mJ\mJ^{\!\top})\right]/d}\;=\;\frac{d}{d+\sigma_{t}^{2}\,\texttt{sg}\!\left[\Tr(\mJ\mJ^{\!\top})\right]},
        \label{eq: trace-weight}
    \end{equation}
    derived below as a regularized BLUE-scalar approximation under the isotropic-noise model. Empirically, the trace weight provides marginal additional gain once the EMA tangent removes the dominant Jacobian amplification, so we keep this construction as an appendix-only Lagrangian implementation of the curvature-budget constraint.

    \paragraph{Setup.} Under the EMA tangent (\Cref{subsec: vamf}), \cref{prop: vamf-grad} and the Reynolds decomposition imply that the per-sample loss conditioned on $\vx_{t}$ admits the noise-decomposition
    \begin{equation}
        \E_{\vx_{0}\mid\vx_{t}}\!\left[\bigl\|V_{\vtheta}^{\text{EMA}}-\vv_{\text{cond}}\bigr\|_{2}^{2}\right]
        \;=\;\underbrace{\bigl\|\tilde{\vr}^{\text{EMA}}\bigr\|_{2}^{2}}_{\text{deterministic signal}}
        \;+\;\underbrace{\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right)}_{\text{irreducible noise floor}\;\nu(\vx_{t},r,t)}.
        \label{eq: blue-decomp}
    \end{equation}
    The noise floor $\nu$ is heteroscedastic: it varies with $(\vx_{t},r,t)$ through both $\mJ$ and $\Sigma_{\vv'}$, and dominates the per-sample loss in high-curvature regions even though it does \emph{not} contribute to the parameter gradient (its $\vtheta$-dependence is wiped by the stop-gradient on the JVP).

    \paragraph{Heteroscedastic-noise BLUE.} Suppose we form a weighted empirical-risk estimator $\hat{\gL}(\vtheta)\!=\!\frac{1}{N}\sum_{i}w_{i}\,\ell_{i}(\vtheta)$ from $N$ independent per-sample losses $\ell_{i}$ whose noise floors $\nu_{i}\!=\!\Var[\ell_{i}\mid\vtheta]$ are heteroscedastic and known. The Best Linear Unbiased Estimator (BLUE) of the population mean assigns
    \begin{equation}
        w_{i}^{\text{BLUE}}\;\propto\;\nu_{i}^{-1},
        \label{eq: blue-classic}
    \end{equation}
    minimizing the variance of $\hat{\gL}$ subject to $\E[\hat{\gL}]\!=\!\E_{\vx_{t},r,t}\!\bigl[\E_{\vx_{0}\mid\vx_{t}}[\ell]\bigr]$. Applying~\eqref{eq: blue-classic} to our setting with $\nu_{i}\!=\!\Tr(\mJ_{i}\Sigma_{\vv',i}\mJ_{i}^{\!\top})$ would prescribe $w\!\propto\!1/\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$. Two practical obstacles remain:
    \begin{enumerate}
        \item The matrix BLUE $w\!\propto\!(\mJ\Sigma_{\vv'}\mJ^{\!\top})^{-1}$ at the gradient level (rather than the trace) requires a per-sample $d\!\times\!d$ inversion, costing $\gO(d^{3})$ per sample.
        \item The scalar form $w\!\propto\!1/\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$ is unbounded as $\Tr(\mJ\mJ^{\!\top})\!\to\!0$ (low-curvature regions), where the BLUE prescribes infinite weight on samples with vanishing noise floor.
    \end{enumerate}

    \paragraph{Isotropic approximation and Tikhonov stabilization.} We resolve both pathologies in two steps. First, the isotropic-noise approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$ collapses the trace to a scalar:
    \begin{equation}
        \Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})
        \;=\;\sigma_{t}^{2}\,\Tr(\mJ\mJ^{\!\top}),
        \label{eq: iso-trace}
    \end{equation}
    which is a single Hutchinson estimate at $\gO(d)$ cost. (\Cref{appx: sigma-schedule} discusses concrete choices of $\sigma_{t}$.) Second, we add a Tikhonov regularizer of magnitude $d$ to the denominator:
    \begin{equation}
        w(\vx_{t},r,t)
        \;:=\;\frac{d}{d+\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})}
        \;=\;\frac{1}{1+\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})/d},
        \label{eq: blue-tikhonov}
    \end{equation}
    matching~\eqref{eq: trace-weight}. The regularization has three desirable properties:
    \begin{itemize}
        \item \textbf{Boundedness.} $w\!\in\!(0,1]$ for all $(\vx_{t},r,t)$, with $w\!=\!1$ exactly when $\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!=\!0$ (no reweighting in flat regions).
        \item \textbf{BLUE limit.} As $\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!\to\!\infty$, $w\!\to\!d/[\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})]$, recovering the BLUE-scalar prescription~\eqref{eq: blue-classic} up to a constant.
        \item \textbf{Bias-variance tradeoff.} The added $d$ trades a small estimator bias (the BLUE limit is approached but not attained) for bounded weights, eliminating the divergence at low curvature.
    \end{itemize}
    The choice of regularizer magnitude $d$ has a clean physical reading: under the BLUE limit the noise contribution per sample equals $w\cdot\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!=\!d$ (i.e., one unit per data dimension), and the regularizer is the smallest constant for which $w\!\le\!1$ uniformly.

    \subsection{Proof of~\Cref{prop: trace-weight} (Trace-Weight Properties)}

    \begin{proposition}[Trace-Weight Properties]
        Under the trace-weight loss~\eqref{eq: vamf-obj} and the isotropic approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$:
        \textbf{(a)} the mean-parameter gradient is the trace-weighted form of~\cref{prop: vamf-grad} with $w$ as a stop-gradient scalar, so the BLUE-style normalization does not enter the parameter update;
        \textbf{(b)} the expected per-sample contribution to the loss decomposes into a deterministic signal $w\,\|\tilde{\vr}^{\text{EMA}}\|_{2}^{2}$ and a constant noise floor $w\,\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!=\!d\,\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})/(d+\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top}))\!\le\!d$, eliminating the unbounded curvature growth from~\cref{thm: jacobi-variance};
        \textbf{(c)} $w$ is monotonically decreasing in $\Tr(\mJ\mJ^{\!\top})$ and, by~\Cref{thm: material-derivative-expansion}, in $(t{-}r)^{2}\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|^{2}$, so high-curvature long-interval samples are automatically downweighted.
        \label[proposition]{prop: trace-weight}
    \end{proposition}

    \begin{proof}
        \textbf{(a) Gradient form.} The trace weight $w$ in~\eqref{eq: trace-weight} is wrapped in a stop-gradient operator on $\Tr(\mJ\mJ^{\!\top})$, so $\nabla_{\vtheta}w\!=\!0$ and $w$ acts as a fixed scalar multiplier. Differentiating the trace-weighted MSE in~\eqref{eq: vamf-obj} therefore gives
        \begin{equation}
            \nabla_{\vtheta}\gL_{\text{VaMF}}^{\text{MF}}\;=\;2\,\E\!\left[w\,(\nabla_{\vtheta}\vu_{\vtheta})^{\!\top}\,\tilde{\vr}^{\text{EMA}}\right]\!,
        \end{equation}
        identical to~\cref{prop: vamf-grad} except that each per-sample contribution carries the scalar weight $w$. The conditional fluctuation $\vv'$ produces a cross term $\E[(\nabla_{\vtheta}\vu_{\vtheta})^{\!\top}\vv']\!=\!\bm{0}$ that vanishes by $\E_{\vx_{0}\mid\vx_{t}}[\vv']\!=\!\bm{0}$, so the BLUE-style normalization does not propagate into the gradient computation.

        \textbf{(b) Bounded noise floor.} Writing $V_{\vtheta}^{\text{EMA}}\!-\!\vv_{\text{cond}}\!=\!\tilde{\vr}^{\text{EMA}}\!-\!\vv'$ with the EMA tangent making $\tilde{\vr}^{\text{EMA}}$ deterministic given $\vx_{t}$, the inner expectation of the per-sample loss is
        \begin{equation}
            \E_{\vx_{0}\mid\vx_{t}}\!\left[w\,\|V_{\vtheta}^{\text{EMA}}\!-\!\vv_{\text{cond}}\|_{2}^{2}\right]\;=\;w\,\|\tilde{\vr}^{\text{EMA}}\|_{2}^{2}+w\cdot\sigma_{t}^{2}\Tr(\Sigma_{\vv'}),
        \end{equation}
        where the cross term vanishes as in (a). Substituting the isotropic-noise approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$ and the explicit form of $w$ yields the noise contribution
        \begin{equation}
            w\cdot\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\;=\;\frac{\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})}{1+\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})/d}\;=\;\frac{d\,\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})}{d+\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})}\;\le\;d,
        \end{equation}
        with equality only in the limit $\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!\to\!\infty$. The trace weight thus caps the noise term at the data dimensionality, eliminating the unbounded curvature growth from~\cref{thm: jacobi-variance}.

        \textbf{(c) Curvature-aware downweighting.} The map $w(\xi)\!=\!1/(1+\sigma_{t}^{2}\xi/d)$ is strictly decreasing in $\xi\!=\!\Tr(\mJ\mJ^{\!\top})\!\ge\!0$. By~\Cref{thm: control-variate-optimum} and the Reynolds decomposition, $\E[\Tr(\mJ\mJ^{\!\top})]$ scales with $(t{-}r)^{2}\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|^{2}$ to leading order, so $w$ shrinks on the same factor and high-curvature long-interval samples are automatically downweighted---realizing the curvature-aware schedule prescribed in~\Cref{appx: blue-scalar} in a single per-sample scalar without requiring an explicit progressive schedule or interval-importance sampling.
    \end{proof}

    \subsection{Acceleration-to-Noise Ratio from the Trace Weight}

    For completeness we record an alternative interpretation of the trace weight in~\cref{prop: trace-weight} as an empirical proxy for the squared \textit{acceleration-to-noise ratio}. This characterization predates the control-variate framing of~\Cref{thm: control-variate-optimum} but remains a useful sanity check in the small-curvature-gap regime; the connection becomes precise when the schedule $\sigma_{t}$ is chosen consistently with the actual conditional-velocity covariance.

    \begin{proposition}[ANR from the Trace Weight]
        Let $\rho(\vx_{t},r,t)\!:=\!\sigma_{t}^{2}\,\Tr(\mJ\mJ^{\!\top})/d$ denote the curvature term in the denominator of~\eqref{eq: trace-weight}, and assume the schedule satisfies $\sigma_{t}^{2}\!=\!\Tr(\Sigma_{\vv'})/d$ (the moment-matched isotropic approximation, achieved exactly by the \textsc{blue-gauss} schedule of~\Cref{appx: sigma-schedule} under Laplace's approximation of $\pdata$). Then near convergence,
        \begin{equation}
            \rho\;\approx\;\frac{\|\Delta(\vx_{t},r,t)\|^{2}}{\Tr(\Sigma_{\vv'})}\;\approx\;\frac{(t-r)^{2}}{4}\cdot\frac{\left\|\frac{\mathrm{D}\vv}{\mathrm{D}t}(\vx_{t},t)\right\|^{2}}{\Tr(\Sigma_{\vv'})}\;=\;\frac{\mathrm{ANR}(\vx_{t},r,t)^{2}}{4},
            \label{eq: anr-trace-weight}
        \end{equation}
        where $\mathrm{ANR}\!:=\!(t-r)\bigl\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\bigr\|/\sqrt{\Tr(\Sigma_{\vv'})}$. The trace weight is then $w\!=\!1/(1+\rho)\!=\!1/(1+\mathrm{ANR}^{2}/4)$.
        \label[proposition]{prop: anr}
    \end{proposition}

    \begin{proof}
        \textbf{Step 1: noise floor and curvature gap.} Under the EMA tangent (\Cref{subsec: vamf}) and at convergence, \cref{prop: bias-var} gives $\tilde{\vr}^{\text{EMA}}\!\approx\!\bm{0}$, so the per-sample loss reduces to its noise floor $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$ (cf.~\eqref{eq: blue-decomp}). Substituting the Reynolds-residual expansion $\vr_{\vtheta}^{\texttt{sg}}\!=\!\Delta+(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu_{\vtheta}$ from~\Cref{subsec: two-roles} and using the MeanFlow identity $(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu\!=\!-\Delta$ gives the per-sample noise floor $\bigl\|\mJ\vv'\bigr\|_{2}^{2}$ at first order; taking expectations and applying the trace identity yields
        \begin{equation}
            \E_{\vx_{0}\mid\vx_{t}}\!\bigl[\|\mJ\vv'\|_{2}^{2}\bigr]
            \;=\;\Tr\!\left(\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\right)
            \;\approx\;\bigl\|\Delta(\vx_{t},r,t)\bigr\|^{2}
            \label{eq: noise-equals-delta}
        \end{equation}
        in the small-gap regime.

        \textbf{Step 2: isotropic moment matching.} Under the schedule assumption $\sigma_{t}^{2}\!=\!\Tr(\Sigma_{\vv'})/d$, the isotropic-noise approximation~\eqref{eq: iso-trace} gives $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})\!=\!\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!=\!\Tr(\Sigma_{\vv'})\Tr(\mJ\mJ^{\!\top})/d$. Combining with~\eqref{eq: noise-equals-delta},
        \begin{equation}
            \frac{\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})}{d}
            \;\approx\;\frac{\|\Delta\|^{2}}{\Tr(\Sigma_{\vv'})}
            \quad\Longleftrightarrow\quad
            \rho\;\approx\;\frac{\|\Delta\|^{2}}{\Tr(\Sigma_{\vv'})}.
            \label{eq: rho-delta-bridge}
        \end{equation}

        \textbf{Step 3: substitute the flow-acceleration expansion.} \Cref{thm: material-derivative-expansion} gives $\|\Delta\|\!\approx\!\tfrac{t-r}{2}\bigl\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\bigr\|$ to leading order, so $\|\Delta\|^{2}/\Tr(\Sigma_{\vv'})\!\approx\!(t-r)^{2}\bigl\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\bigr\|^{2}/[4\,\Tr(\Sigma_{\vv'})]\!=\!\mathrm{ANR}^{2}/4$, completing~\eqref{eq: anr-trace-weight}.
    \end{proof}

    \paragraph{Schedule sensitivity.} The proof requires the moment-matched schedule $\sigma_{t}^{2}\!=\!\Tr(\Sigma_{\vv'})/d$. Practical schedules (\Cref{appx: sigma-schedule}) only approximate this:
    \begin{itemize}
        \item \textsc{blue-gauss} matches it exactly under Laplace's approximation of $\pdata$, so~\eqref{eq: anr-trace-weight} holds to leading order.
        \item \textsc{u-shape} qualitatively tracks the location of the peak of $\Tr(\Sigma_{\vv'})$ (\cref{eq: gauss-sigma-supp}) but not its absolute scale, so the connection holds up to a $t$-dependent multiplicative constant.
        \item \textsc{linear} and \textsc{quadratic} are monotone in $t$ and \emph{do not} approximate the U-shape; the connection in~\eqref{eq: anr-trace-weight} is then valid only as a qualitative trend rather than a quantitative identity.
    \end{itemize}
    The trace weight remains a valid BLUE-scalar approximation in all three cases (\Cref{appx: blue-scalar}); only the ANR \emph{interpretation} of $\rho$ is schedule-sensitive.

    \subsection{Schedule $\sigma_{t}$ in the Trace Weight}
    \label{appx: sigma-schedule}

    The trace weight~\eqref{eq: trace-weight} introduces a single hyperparameter, the time-dependent scalar $\sigma_{t}$ that approximates the conditional-velocity standard-deviation scale via $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$. The exact $\Sigma_{\vv'}$ depends on $\pdata$ and is not generally closed-form; we first derive a tractable instance under Laplace's approximation $\pdata\!\approx\!\gN(\bm{0},\sigma_{d}^{2}\mI_{d})$, then compare five practical schedules empirically.

    \paragraph{Conditional-velocity variance under Laplace's approximation.} Following Laplace's approximation in Bayesian statistics~\cite{tierney1986accurate,mackay1992practical}, we replace $\pdata$ by a Gaussian centered at the mode with covariance set to the inverse Hessian of $-\log\pdata$ at that mode. Take $\vx_{0}\!\sim\!\gN(\bm{0},\sigma_{d}^{2}\mI_{d})$ and $\vx_{1}\!\sim\!\gN(\bm{0},\mI_{d})$ with the linear interpolant $\vx_{t}\!=\!(1-t)\vx_{0}+t\vx_{1}$. Conditional on $\vx_{0}$, the noise contribution is $\vx_{t}\mid\vx_{0}\!\sim\!\gN((1-t)\vx_{0},t^{2}\mI_{d})$. Combining this Gaussian likelihood with the Gaussian prior on $\vx_{0}$ via conjugate inference,
    \begin{equation}
        \mathrm{Var}\!\left[\vx_{0}\mid\vx_{t}\right]
        \;=\;\frac{t^{2}\sigma_{d}^{2}}{\sigma_{d}^{2}(1-t)^{2}+t^{2}}\,\mI_{d}.
    \end{equation}
    Substituting $\vv_{\text{cond}}=\vx_{1}-\vx_{0}=(\vx_{t}-\vx_{0})/t$ gives
    \begin{equation}
        \Sigma_{\vv'}(\vx_{t},t)
        \;=\;\frac{1}{t^{2}}\,\mathrm{Var}[\vx_{0}\mid\vx_{t}]
        \;=\;\frac{\sigma_{d}^{2}}{\sigma_{d}^{2}(1-t)^{2}+t^{2}}\,\mI_{d}.
        \label{eq: gauss-sigma-supp}
    \end{equation}
    \Cref{eq: gauss-sigma-supp} is U-shaped in $t$: at $t=0$ and $t=1$ it equals $\sigma_{d}^{2}$ (resp.\ $1$ for the unit-variance case $\sigma_{d}=1$), and at $t=0.5$ it reaches its maximum of $2\sigma_{d}^{2}/(\sigma_{d}^{2}+1)$. The peak coincides exactly with the regime where the Jacobian-deviation norm $\Tr(\mJ\mJ^{\!\top})$ from~\cref{thm: jacobi-variance} is largest, since both reflect the same physical phenomenon---mid-$t$ mode mixing.

    \paragraph{Practical schedules.} Real $\pdata$ (e.g.\ image latents) is non-Gaussian and Eq.~\eqref{eq: gauss-sigma-supp} is at best a first-order approximation. We therefore benchmark five practical schedules:

    \begin{description}
        \item[\textsc{none}.] $\sigma_{t}\!=\!1$. The bare BLUE-scalar form. Serves as the cleanest test of the curvature term in isolation.
        \item[\textsc{linear}.] $\sigma_{t}\!=\!t$. The simplest monotone schedule with $\sigma_{t}^{2}\!=\!t^{2}$ in the denominator of~\eqref{eq: trace-weight}. Suppresses high-$t$ samples where the noise has dispersed.
        \item[\textsc{quadratic}.] $\sigma_{t}\!=\!t^{2}$, so $\sigma_{t}^{2}\!=\!t^{4}$. Steeper monotone profile, used in early experiments.
        \item[\textsc{u-shape}.] $\sigma_{t}\!=\!\sqrt{t(1-t)}$, so $\sigma_{t}^{2}\!=\!t(1-t)$. Peaks at $t=0.5$ and vanishes at the endpoints---qualitatively matching the location of~\eqref{eq: gauss-sigma-supp}'s peak without committing to a parametric form.
        \item[\textsc{blue-gauss}.] $\sigma_{t}^{2}\!=\!1/(\sigma_{d}^{2}(1-t)^{2}+t^{2})$ from~\eqref{eq: gauss-sigma-supp} with $\sigma_{d}\!=\!1$. The closed-form BLUE under Laplace's approximation of $\pdata$.
    \end{description}

    The toy benchmark sweep across all five schedules appears in~\Cref{sec: experiment}, the toy schedule sweep (\Cref{appx: sigma-schedule}); the empirical winner ($\sigma_t\!=\!1$) is used as the default in the main text and in~\Cref{appx: dit-results}.
