\section{Preliminaries}
\label{sec: preliminary}

\paragraph{Flow-Matching.} Let $\vx$ denote a data sample in $\gX\subset\mathbb{R}^{d}$ . Flow-matching generative models~\cite{lipman2023flowmatchinggenerativemodeling,tong2024improvinggeneralizingflowbasedgenerative} learn a time-dependent diffeomorphism $\psi(\vx,t):\mathbb{R}^{d}\!\times\![0,1]\to\mathbb{R}^{d}$ from a simple distribution $\rvx_{1}\!\sim\!{p}(\rvx,1)$ to the target $\rvx_{0}\!\sim\!{p}(\rvx,0)\!=\!\pdata$ by parameterizing a velocity field $\vv(\vx,t)\!\triangleq\!\frac{\mathrm{d}}{\mathrm{d}t}\psi(\vx,t)$. Since the exact \textit{marginal velocity field} $\vv(\vx,t)$ is generally inaccessible, conditional flow matching instead learns a conditional field $\vv(\vx,t\mid\vx_{0})\!=\!\vx_{1}-\vx_{0}$ with $\vx_{t}\!=\!(1-t)\vx_{0}+t\vx_{1}$. The following lemma justifies the substitution. See~\Cref{appx: first} for its proof.
\begin{lemma}
    Given vector fields $\vv(\vx,t\mid\vx_{0})$ generating probability paths $p(\vx,t\mid\vx_{0})$, for any $\rvx_{0}\!\sim\!p(\rvx_{0})$, the expected conditional velocity field is equal to the marginal velocity field
    \begin{equation}
        \vv(\vx,t)=\E_{\vx_{0}\sim{p(\vx_{0},t\mid\vx)}}\!\left[\vv(\vx,t\mid\vx_{0})\right].
        \label{eq: margv-condv-relation}
    \end{equation}
\end{lemma}

\paragraph{One-step Mean Flows.} Iterative integration of $\vv(\vx,t)$ at inference is expensive and can be numerically unstable. To bypass it, Mean Flows~\cite{geng2025meanflowsonestepgenerative} learn a two-parameter average velocity field $\vu(\rvx,r,t)$ through regressing the identity
\begin{equation}
    (t-r)\,\vu(\vx,r,t)=\int_{r}^{t}\vv(\vx,\tau)\,\mathrm{d}\tau.
    \label{eq: meanflow-identity}
\end{equation}
Differentiating with respect to the terminal time $t$ yields
\begin{equation}
    \vu(\vx,r,t)+(t-r)\tfrac{\mathrm{d}}{\mathrm{d}t}\vu(\vx,r,t)=\vv(\vx,t),\qquad
    \tfrac{\mathrm{d}}{\mathrm{d}t}\vu=\partial_{\vx}\vu\!\cdot\!\vv(\vx,t)+\partial_{t}\vu,
    \label{eq: td-meanflow-identity}
\end{equation}
where the total derivative $\frac{\mathrm{d}}{\mathrm{d}t}\vu$ is essentially a JVP, denoted by $\texttt{JVP}(\vu,(\vx,r,t),(\vv(\vx,t),0,1))$. The original MeanFlow paper~\cite{geng2025meanflowsonestepgenerative} replaces \emph{both} occurrences of the marginal field with the conditional field $\vv_{\text{cond}}\!=\!\vx_{1}\!-\!\vx_{0}$, giving the MeanFlow loss
\begin{equation}
    \mathcal{L}_{\text{MF}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\!\left[\left\|\vu_{\vtheta}+(t-r)\,\texttt{sg}\!\left[\texttt{JVP}(\vu_{\vtheta},(\vz,r,t),(\vv_{\text{cond}},0,1))\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: mf-loss}
\end{equation}
with $r,t\!\sim\!\gU(0,1)$, $r\!\leq\!t$, $\vx_{0}\!\sim\!\pdata$, $\vx_{1}\!\sim\!\gN(0,\mI_{d})$, $\vz\!=\!(1-t)\vx_{0}+t\vx_{1}$, and the stop-gradient $\texttt{sg}[\cdot]$ avoiding double backpropagation through the JVP.

\section{Variance Reduction in Learning Mean Flows}
\label{sec: method}

The MeanFlow loss in~\eqref{eq: mf-loss} is empirically known to be \textit{non-decreasing} and to suffer from \textit{high-variance gradients}~\cite{zhang2025alphaflowunderstandingimprovingmeanflow,geng2025improvedmeanflowschallenges}. We investigate the statistical mechanism behind this pathology. Importantly, our analysis isolates the role of the conditional velocity field $\vv_{\text{cond}}$ when it appears as the JVP tangent, and identifies that it acts as a Monte Carlo control variate for the inaccessible marginal velocity, with a coefficient that the original MeanFlow fixes to a suboptimal value. We derive the optimal coefficient and establish a connection to practical fixes in concurrent works.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Two roles of the conditional velocity field}
\label{subsec: two-roles}

With Reynolds decomposition, we express the conditional velocity by $\vv_{\text{cond}}\!=\!\vv(\vx_{t},t)+\vv'$ with $\E_{\vx_{0}\mid\vx_{t}}[\vv']\!=\!\bm{0}$ according to~\eqref{eq: margv-condv-relation} and $\Sigma_{\vv'}\!\triangleq\!\Cov_{\vx_{0}\mid\vx_{t}}[\vv']$. The MeanFlow loss in~\eqref{eq: mf-loss} substitutes $\vv_{\text{cond}}$ in two distinct positions: as the \textit{regression target} $-\vv_{\text{cond}}$, and as the \textit{tangent} inside the total derivative. Carrying the decomposition through per-sample loss $\ell_{\text{MF}}(\vtheta)$ yields:
\begin{equation}
    \E_{\vx_{0}\mid\vx_{t}}\!\bigl[\ell_{\text{MF}}(\vtheta)\bigr]
    =\bigl\|\vr_{\vtheta}^{\texttt{sg}}\bigr\|^{2}_{2}
    +\texttt{sg}\!\bigl[\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})\bigr],
    \label{eq: per-sample-expand}
\end{equation}
where $\mJ\!\triangleq\!(t{-}r)\partial_{\vz}\vu_{\vtheta}\!-\!\mI_{d}\!\in\!\mathbb{R}^{d\times d}$ is a \textit{Jacobi factor} determined by the Jacobian matrix $\partial_{\vz}\vu_{\vtheta}$ and $\vr_{\vtheta}^{\texttt{sg}}\!\triangleq\!\vu_{\vtheta}\!+\!(t{-}r)\,\texttt{sg}[\partial_{\vz}\vu_{\vtheta}\!\cdot\!\vv\!+\!\partial_{t}\vu_{\vtheta}]\!-\!\vv$ is the deterministic mean-field residual. Since the residual $\vr_{\vtheta}^{\texttt{sg}}$ is the exact loss we aim to minimize given by~\eqref{eq: td-meanflow-identity}, the trace term leads to the non-decreasing loss and eliminates $\textit{iff}$ the Jacobi satisfies $\partial_{\vx}\vu_{\vtheta}=(t-r)\mI_{d}$. As a result, we identify two roles of the conditional velocity field in this decomposition:

\noindent\textbf{Unbiased Target.} As the regression target, $\vv_{\text{cond}}$ injects $\vv'$ \textit{linearly} into the residual, contributing the \emph{irreducible} per-sample noise floor $\Tr(\Sigma_{\vv'})$. This contribution is independent of $\vtheta$ and reflects the natural Monte Carlo cost of replacing the inaccessible $\vv$ by a per-sample proxy.

\noindent\textbf{Variance Amplifier.} As the JVP tangent, the same $\vv_{\text{cond}}$ injects $\vv'$ \textit{multiplicatively} through the Jacobi factor $\mJ$, contributing the \emph{amplified} term $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$. It scales unboundedly with the spectral magnitude of $\mJ$. We identify this amplification as the dominant driver of variance at the gradient level:

\begin{theorem}[Jacobian Variance Amplification]
    Let $\boldsymbol{g}\!\triangleq\!\nabla_{\vtheta}\vu_{\vtheta}(\vx_{t},r,t)\!\in\!\mathbb{R}^{d\times p}$ be the gradient of the average velocity. The conditional per-step gradient variance of $\ell_{\text{MF}}(\vtheta)$ in~\eqref{eq: mf-loss} satisfies
    \begin{equation*}
        \boxed{
            \Var\!\left[\nabla_{\vtheta}\ell_{\text{MF}}\mid\vx_{t}\right]
            \;\propto\;
            \boldsymbol{g}^{\!\top}\,\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\,\boldsymbol{g}.
        }
    \end{equation*}
    \label[theorem]{thm: jacobi-variance}
\end{theorem}

See the proof in~\Cref{appx: theorem-proof}. Our \Cref{thm: jacobi-variance} identifies that the gradient variance grows with $\|\mJ\|^{2}$, saturates only at the irreducible noise floor when $\mJ\!\to\!\bm{0}$, and is \emph{uncontrolled} by any term in the loss due to the use of the stop-gradient in the original MeanFlow. We notice that, without the stop-gradient operator, the optimizer could in principle reduce variance by driving $\mJ\!\to\!\bm{0}$. This establishes a \textit{semi-gradient gap}, formally,

\begin{theorem}[Semi-Gradient Gap]
    The gradient of the MeanFlow loss $\gL_{\text{MF}}$ with and without the stop-gradient operator differ by
    \begin{equation*}
        \boxed{
            \underbrace{2(t{-}r)\,\E\!\left[\nabla_{\vtheta}\!\left(\partial_{\vx_{t}}\vu_{\vtheta}\!\cdot\!\vv+\partial_{t}\vu_{\vtheta}\right)\!\cdot\!\vr_{\vtheta}\right]}_{\text{mean-field gradient difference}}
            +\underbrace{\E\!\left[\nabla_{\vtheta}\Tr\!\left(\mJ\Sigma_{\vv'}\mJ^{\!\top}\right)\right]}_{\text{variance-driven gradient difference}}.
        }
    \end{equation*}
    \label[theorem]{thm: semi-gradient-gap}
\end{theorem}

At convergence the mean-field difference vanishes ($\vr_{\vtheta}\!\to\!\bm{0}$) and the variance-driven term $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$ dominates, but the stop-gradient hides it from the optimizer---producing the empirical non-decreasing-loss pathology. \Cref{fig: cm-gap} visualizes the per-state RMS fluctuation $\sqrt{\Tr(\Sigma_{\vv'}\!\mid\!\vx_{t})}$ on a 2-D Gaussian mixture: the magnitude is non-zero, concentrates in mode-mixing regions where conditional paths overlap, and grows toward the latent endpoint.

\subsection{Rethinking tangent as a control variate}
\label{subsec: control-variate}

The amplification in~\Cref{thm: jacobi-variance} arises due to using $\vv_{\text{cond}}\!=\!\vv\!+\!\vv'$ as the JVP tangent injects the zero-mean fluctuation $\vv'$. This actually corresponds to the canonical structure of a Monte Carlo \textit{control variate}~\cite{glasserman2003montecarlo}, where $\vv_{\text{cond}}$ is a stochastic estimator of the inaccessible $\vv$ with known mean, and the coefficient on the noise term $\vv'$ can be a free parameter that determines the bias-variance trade-off. We make this explicit by introducing a tangent-mixing coefficient $\beta\!\in\![0,1]$:
\begin{equation}
    \tilde{\vv}_{\text{tang}}(\beta)\;\triangleq\;(1-\beta)\,\vv_{\text{cond}} + \beta\,\hat{\vv}\;=\;\vv + (1-\beta)\,\vv' + \beta\,\vb,
    \label{eq: tangent-mix}
\end{equation}
where $\hat{\vv}$ can be any deterministic proxy for the marginal velocity (\textit{e.g., a velocity network}) with bias $\vb\!\triangleq\!\hat{\vv}-\vv$ that is deterministic given $\vx_{t}$. Herein, $\beta\!=\!0$ recovers vanilla MeanFlow and $\beta\!=\!1$ replaces the tangent entirely with the deterministic proxy.

\noindent\textbf{Bias-variance trade-off.} Substituting $\tilde{\vv}_{\text{tang}}(\beta)$ for $\vv_{\text{cond}}$ in the JVP tangent and propagating through the loss, the per-sample gradient writes
\begin{equation}
    \boldsymbol{g}^{(\beta)} \;=\; 2\,\boldsymbol{g}^{\!\top}\!\Bigl[\vr_{\vtheta} \;+\; \bigl((1{-}\beta)\,\mJ - \beta\,\mI\bigr)\,\vv' \;+\; \beta\,(\mJ{+}\mI)\,\vb\Bigr],
    \label{eq: g-beta}
\end{equation}
where the noise term scales the conditional fluctuation $\vv'$ by the matrix $((1{-}\beta)\,\mJ - \beta\,\mI)$, and the bias term scales the proxy error $\vb$ by $\beta\,(\mJ{+}\mI)$. This establishes the following theorem.

\begin{theorem}[Optimal control-variate coefficient]
    Define the \emph{conditional MSE} of the per-sample gradient~\eqref{eq: g-beta} as its mean-squared error against its noise- and bias-free target $2\,\boldsymbol{g}^{\!\top}\vr_{\vtheta}$, taken over the randomness of $\vv'$ at fixed $\vx_{t}$:
    \begin{equation*}
        M(\beta)\;\triangleq\;\E_{\vv'\mid\vx_{t}}\bigl[\bigl\|\boldsymbol{g}^{(\beta)}-2\,\boldsymbol{g}^{\!\top}\vr_{\vtheta}\bigr\|_{2}^{2}\bigr].
    \end{equation*}
    Under the scalar-isotropic approximation $\Sigma_{\vv'}\!\approx\!\sigma^{2}\mI_{d}$ and $\mJ\!\approx\!\kappa\mI_{d}$, $M(\beta) = \beta^{2}(\kappa{+}1)^{2}\|\vb\|^{2} + \sigma^{2}d\bigl((1{-}\beta)\kappa - \beta\bigr)^{2}$ and admits a unique minimizer
    \begin{equation*}
        \boxed{
            \beta^{\star} \;=\; \underbrace{\frac{\kappa}{\kappa+1}}_{\text{noise-cancellation}}\;\cdot\;\underbrace{\frac{\sigma^{2}d}{\sigma^{2}d + \|\vb\|^{2}}}_{\text{shrinkage}},
        }
    \end{equation*}
    with optimum value $M(\beta^{\star})\!=\!\sigma^{2}d\,\kappa^{2}\,\|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2})$. For all $\kappa,\|\vb\|^{2}\!>\!0$, $M(\beta^{\star})$ strictly dominates both corner MSEs $M(0)\!=\!\sigma^{2}\kappa^{2}d$ (vanilla MeanFlow) and $M(1)\!=\!(\kappa{+}1)^{2}\|\vb\|^{2}+\sigma^{2}d$ (deterministic-tangent estimator with bias).
    \label[theorem]{thm: control-variate-optimum}
\end{theorem}

The proof is a direct application of bias-variance decomposition to~\eqref{eq: g-beta} and is given in~\Cref{appx: theorem-proof}. The two factors of $\beta^{\star}$ have crisp interpretations: $\kappa/(\kappa{+}1)$ is the coefficient that \emph{cancels the Jacobian-amplified noise} via destructive interference between the $\mJ\vv'$ and $\mI\vv'$ contributions in~\eqref{eq: g-beta}; $\sigma^{2}d/(\sigma^{2}d{+}\|\vb\|^{2})$ is the standard James-Stein shrinkage that pulls $\beta^{\star}$ toward the unbiased corner when the proxy bias $\|\vb\|$ is large. As the proxy bias shrinks ($\|\vb\|^{2}\!\to\!0$), $\beta^{\star}\!\to\!\kappa/(\kappa{+}1)$, and as the Jacobian magnitude grows ($\kappa\!\to\!\infty$), $\beta^{\star}\!\to\!1$.

\paragraph{Vanilla MeanFlow is generally suboptimal.} Vanilla MeanFlow uses $\beta\!=\!0$ which is optimal only when $\kappa\!=\!0$ (i.e., $\partial_{\vz}\vu_{\vtheta}\!=\!\frac{1}{t-r}\mI_{d}$) or when no deterministic proxy is available ($\|\vb\|\!=\!\infty$). Both conditions fail in practice: trained backbones produce $\partial_{\vz}\vu_{\vtheta}$ matrices that are spectrally far from $\frac{1}{t-r}\mI_{d}$, and an EMA copy of $\vu_{\vtheta}$ provides a viable deterministic proxy at near-zero compute cost.

\paragraph{Concurrent fixes through this lens.} Several remedies for MeanFlow's training pathology in concurrent work can now be located in the same one-parameter family. Each fix corresponds to a different practical realization of the optimum, distinguished by \emph{which proxy} they use for $\vv$ and \emph{which value of $\beta$} the construction implicitly selects:
\begin{itemize}
    \item \textbf{Deterministic-tangent methods} (Improved MF~\cite{geng2025improvedmeanflowschallenges}, EMA tangents) take $\beta\!\to\!1$ with a learned or EMA-derived proxy $\hat{\vv}$. The amplified noise term is eliminated; the bias term $(\mJ{+}\mI)\vb$ is controlled by training the proxy.
    \item \textbf{Reparameterization methods} (TVM~\cite{zhou2026terminalvelocitymatching}) sidestep the spatial Jacobian by reparameterizing the total derivative, effectively setting $\mJ\!=\!\bm{0}$ in our framework.
    \item \textbf{Curvature-reduction methods} (Re-MeanFlow~\cite{zhang2026overcomingcurvaturebottleneckmeanflow}) reduce $\|\mJ\|$ via reflow preprocessing, lowering $\kappa$ and the amplified-noise term.
    \item \textbf{Curriculum methods} (AlphaFlow~\cite{zhang2025alphaflowunderstandingimprovingmeanflow}) avoid high-$\kappa$ regimes during training rather than reducing the variance directly.
\end{itemize}
None of these methods explicitly identify the control-variate structure or derive $\beta^{\star}$. We do, and we use the result both as an analytical lens for these methods and as the foundation for our practical training framework.

\subsection{Empirical instantiation: Variance-Aware MeanFlow (VaMF)}
\label{subsec: vamf}

To validate the framework empirically, we need a concrete realization of $\beta^{\star}$ that we can sweep at scale. We construct \textit{Variance-Aware MeanFlow} (VaMF) as a controlled instance of the $\beta\!=\!1$ corner: the JVP tangent is replaced by an EMA-derived deterministic proxy $\hat{\vv}\!=\!\vu_{\bar{\vtheta}}(\vx_{t},t,t)$, and a small flow-matching anchor at $r\!=\!t$ keeps the proxy bias $\vb\!=\!\hat{\vv}-\vv$ small so the bias term $\beta(\mJ{+}\mI)\vb$ in~\eqref{eq: g-beta} stays controlled. The regression target remains $\vv_{\text{cond}}$ unchanged---only the tangent is replaced, exploiting the role asymmetry of~\Cref{subsec: two-roles}. The construction adds one extra JVP per step relative to vanilla MeanFlow. \Cref{appx: implementation} gives the full loss, training algorithm, and three propositions characterizing the resulting gradient form, residual bias, and the necessity of the target--tangent role split. In~\Cref{sec: experiment} we sweep $\beta\!\in\!\{0,0.25,0.5,1\}$ with VaMF realizing the $\beta\!=\!1$ corner, the linear interpolation realizing $\beta\!\in\!\{0.25,0.5\}$, and vanilla MeanFlow realizing $\beta\!=\!0$.
