\section{Preliminary}
\label{sec: preliminary}

\paragraph{Flow-Matching.} Let $\gX\subset\mathbb{R}^{d}$ be the space of random variables $\rvx\in\mathbb{R}^{d}$ and $\vx$ a data sample as the realized state. The objective of flow-matching generative models is to learn a time-dependent diffeomorphic map $\psi(\vx,t):\mathbb{R}^{d}\times[0,1]\to\mathbb{R}^{d}$ from a simple distribution $\rvx_{1}\sim{p}(\rvx,1)$ to the target data distribution $\rvx_{0}\sim{p}(\rvx,0)={\pdata}$. Specifically, this is achieved by learning a \textit{time-dependent velocity field} $\vv(\vx,t):\mathbb{R}^{d}\times[0,1]\to\mathbb{R}^{d}$:
\begin{equation}
    \vv(\vx, t)\triangleq{\frac{\mathrm{d}}{\mathrm{d}t}\psi(\vx,t)},\quad \psi(\vx,0)=\vx_{0}.
    \label{eq: velocity-field}
\end{equation}
The map $\psi(\vx,t)$ induces a \textit{time-dependent probability density path} $p(\vx, t):\mathbb{R}^{d}\times[0,1]\to\mathbb{R}_{>0}$, with $\int_{\gX}p(\vx,t)\mathrm{d}\vx$, $\forall{t}\in[0,1]$.
% through the pushforward measure $p(\vx,t)\triangleq{\left[\psi\right]_{\#}(p(\vx,1))}$, where the pushforward operator $[\cdot]_{\#}$ is defined by the change-of-variable operation
% \begin{equation}
%     \left[\psi\right]_{\#}(p(\vx,1))=p(\psi^{-1}(\vx,t))\left|\det\left[\frac{\partial}{\partial\vx}\psi^{-1}(\vx,t)\right]\right|.
%     \label{eq: pushforward}
% \end{equation}
A sufficient and necessary condition for a vector field to generate a probability path is the \textit{continuity equation}, which is a partial differential equation (PDE) that writes
\begin{equation}
    \frac{\partial}{\partial t}p(\vx,t)+
    \text{div}\left(p(\vx,t)\vv(\vx,t)\right)=0,
\end{equation}
where $\text{div}(f)$ denotes the divergence of function $f$. However, the exact \textit{marginal velocity field} $\vv(\vx,t)$ is generally \textit{inaccessible}, especially in a high-dimensional sample space. The optimal-transport conditional flow-matching (OT-CFM)~\cite{lipman2023flowmatchinggenerativemodeling,tong2024improvinggeneralizingflowbasedgenerative} alternatively learns a conditional velocity field $\vv(\vx,t\mid{\vx_{0};\vtheta})\triangleq{\vv\left((1-t)\vx_{0}+t\vx_{1};\vtheta\right)}$
% by minimizing a loss function:
% \begin{equation}
%     \mathcal{L}_{\text{FM}}(\vtheta)=
%     \E_{t\sim\gU(0,1),\vx_{0}\sim{\pdata},\vx_{1}\sim\gN(0,\mI_{d})}
%     \left\|\vv\left(\vx,t;\vtheta\right)-(\vx_{1}-\vx_{0})\right\|_{2}^{2}.
%     \label{eq: cfm-obj}
% \end{equation}
% Herein, $t\sim\gU(0,1)$ is a scalar sample from the uniform distribution defined on the range of $[0,1]$ and $\vx_{1}\sim\gN(0,\mI_{d})$ is a white noise sample from the standard normal distribution defined in $\mathbb{R}^{d}$. Meanwhile,
The marginal velocity field equals the expected conditional velocity field $\vv(\vx,t)\!=\!\E_{\vx_{0}\sim p(\vx_{0},t\mid\vx)}[\vv(\vx,t\mid\vx_{0})]$ \eqref{eq: margv-condv-relation}, which justifies the surrogate-loss substitution in CFM (full statement and proof in~\Cref{appx: first}).
\begin{equation}
    \vv(\vx,t)=\E_{\vx_{0}\sim{p(\vx_0,t\mid{\vx})}}\!\left[\vv(\vx,t\mid{\vx_{0}})\right]
    \label{eq: margv-condv-relation}
\end{equation}

\paragraph{One-step Mean Flows.} The issue with diffusion~\cite{ho2020denoising,song2021scorebased} and flow-matching models~\cite{lipman2023flowmatchinggenerativemodeling} is the integration. Evaluating the velocity field multiple times is not only expensive but can also lead to numerical instability. A Mean Flow model~\cite{geng2025meanflowsonestepgenerative} instead learns a two-parameter average velocity field $\vu(\rvx,r,t)$ to enable efficient sampling. It leverages the identity between the average and marginal velocity field given by
\begin{equation}
    (t-r)\vu(\vx,r,t)=\int_{\tau=r}^{t}\vv(\vx,\tau)\mathrm{d}\tau,
    \label{eq: meanflow-identity}
\end{equation}
where $r$ and $t$ are scalars, the start and end time steps along the probability density path. Taking the total derivative with respect to the terminal time step $t$ yields
\begin{equation}
    \vu(\vx,r,t)+(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu(\vx,r,t)=\vv(\vx,t),
    \label{eq: td-meanflow-identity}
\end{equation}
where the total derivative of the average velocity field with respect to $t$ can be expressed as a Jacobi-vector Product (JVP)
\begin{equation}
    \frac{\mathrm{d}}{\mathrm{d}t}\vu(\vx,r,t)
    =\partial_{\vx}\vu\cdot \vv(\vx,t)
    +\partial_{r}\vu\cdot\frac{\mathrm{d}r}{\mathrm{d}t}
    +\partial_{t}\vu\cdot\frac{\mathrm{d}t}{\mathrm{d}t}
    =\partial_{\vx}\vu\cdot \vv(\vx,t)+\partial_{t}\vu
    \label{eq: dudt}
\end{equation}

We abuse the notation and express the right-hand side by $\texttt{JVP}(\vu,(\vx,r,t),(\vv(\vx,t),0,1))$. In the original MeanFlow~\cite{geng2025meanflowsonestepgenerative}, the authors replace both the marginal velocity field in the right-hand side of~\eqref{eq: td-meanflow-identity} and inside the JVP with the conditional velocity field, and derive the MeanFlow loss function given by
\begin{equation}
    \mathcal{L}_{\text{MF}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+
    (t-r)\texttt{sg}\left[\texttt{JVP}(\vu_{\vtheta},(\vz,r,t),(\vv_{\text{cond}},0,1))\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: mf-loss}
\end{equation}
where $r\sim\gU(0,1)$, $t\sim\gU(0,1)$, $r\leq{t}$, $\vx_{0}\sim\pdata$, $\vx_{1}\sim\gN(0,\mI_{d})$, $\vz=(1-t)\vx_{0}+t\vx_{1}$, $\vv_{\text{cond}}=\vx_{1}-\vx_{0}$, and $\texttt{sg}[\cdot]$ is the stop-gradient operator. The authors highlight that stopping gradients in JVP helps eliminate the need for ``double backpropagation'' and high-order derivatives.

\section{Variance-aware Mean Flows}
\label{sec: method}

The MeanFlow loss in~\eqref{eq: mf-loss} is empirically known to be \textit{non-decreasing} and to suffer from \textit{high-variance gradients}~\cite{zhang2025alphaflowunderstandingimprovingmeanflow,geng2025improvedmeanflowschallenges}. In this section, we aim to answer the two complementary questions:

    rightline=false,
    bottomline=false,
    backgroundcolor=dust!15
]
\noindent\textbf{\textsl{RQ1} (Diagnosis):} What is the root cause non-decreasing, high-variance MeanFlow loss?
\end{mdframed}

\begin{mdframed}[%

    linecolor=teal,

    topline=false,
    rightline=false,
    bottomline=false,
    backgroundcolor=dust!15
]
\noindent\textbf{\textsl{RQ2} (Prescription):} Given the identified cause, what is the simplest modification that drives the gradient variance toward its irreducible level?
\end{mdframed}

In the following,~\cref{subsec: mf-bias-vars} answers~\textsl{RQ1} at the level of the gradients. We apply Reynolds decomposition to reveal both the bias induced by stop-gradient training and the per-step gradient variance governed by the Jacobi factor $\mJ=(t-r)\partial_{\vz}\vu_{\vtheta}-\mI_{d}$, and establish a connection between the bias and the \textit{curvature gap} $\Delta=\vu-\vv$. \Cref{subsec: flow-acc} then quantifies the gap in terms of flow acceleration with a Taylor expansion, exposing the material derivative as its physical source of the curvature gap. \Cref{subsec: vamf} answers~\textsl{RQ2} by proposing our training framework, \textit{Variance-Aware MeanFlow (VaMF)}, comprising an EMA velocity anchor, a flow-matching anchor, and a closed-form trace weight, all applied on top of the original MSE objective without any architectural change. Full proofs are deferred to~\Cref{appx: theorem-proof}.

\subsection{Bias and Variances in Learning Mean Flows}
\label{subsec: mf-bias-vars}

We begin with investigating the consequences of replacing the inaccessible marginal velocity $\vv(\vx,t)$ by the conditional field $\vv_{\text{cond}}=\vx_{1}-\vx_{0}$ in both the JVP tangent and the regression target. We first apply the Reynolds decomposition of the field and let $\vv(\vx,t\mid{\vx_{0}})=\vv(\vx,t)+\vv^{\prime}$, where $\vv^{\prime}$ is a zero-mean fluctuation from the marginal velocity induced by optimal transport between latent $\vx_{1}$ and data $\vx_{0}$. The regression loss~\eqref{eq: mf-loss} then expands to
\begin{equation}
    \begin{aligned}
        \gL_{\text{MF}}(\vtheta)
        &=\E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+(t-r)\texttt{sg}\left[\partial_{\vz}\vu_{\vtheta}\cdot\vv_{\text{cond}}+\partial_{t}\vu_{\vtheta}\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right] \\
        % &=\E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+(t-r)\texttt{sg}\left[\partial_{\vz}\vu_{\vtheta}\cdot \vv(\vx_{t},t)+\partial_{t}\vu_{\vtheta}\right]-\vv(\vx_{t},t)+\texttt{sg}[\mJ]\vv^{\prime}\right\|_{2}^{2}\right]\\
        &=\E_{r,t,\vx_{t},\vx_{1}}\left[\E_{\vx_{0}\mid{\vx_{t}}}\left[\left\|\underbrace{\vu_{\vtheta}+(t-r)\texttt{sg}\left[\partial_{\vz}\vu_{\vtheta}\cdot \vv(\vx_{t},t)+\partial_{t}\vu_{\vtheta}\right]-\vv(\vx_{t},t)}_{\text{mean-field residual }\vr^{\texttt{sg}}_{\vtheta}(\vx_{t},t)}+\texttt{sg}[\mJ]\vv^{\prime}\right\|_{2}^{2}\right]\right].
    \end{aligned}
    \label{eq: expanded-mf-loss}
\end{equation}

\begin{figure}[t]
    \centering
    \includegraphics[width=\textwidth]{assets/figures/illustration.pdf}
    \caption{Conditional-marginal velocity gap on a 3-mode mixture. \textbf{Top:} the expected per-sample fluctuation $\E_{\vx_{0}\mid\vx_{t}}\|\vv(\vx_{t},t) - \vv(\vx_{t},t\mid\vx_{0})\|$, evaluated along the conditional paths, grows monotonically with $t$ as conditional trajectories disperse, with the steepest rise where $\vx_{t}$ traverses the mode-mixing region. \textbf{Bottom:} the spatial heatmap at three timesteps confirms that the gap is heterogeneous and concentrates in the mode-mixing regions where multiple conditional paths overlap.}
    \label{fig: cm-gap}
    \vspace{-1em}
\end{figure}

Herein, we abuse the notations and let $\mJ\triangleq(t-r)\partial_{\vz}\vu_{\theta}-\mI_{d}\in\mathbb{R}^{d\times d}$ denote a Jacobi factor of $\vu_{\vtheta}$ given sample $\vx_{t}=\vx_{1}-\vx_{0}$. This decomposition reveals that the original loss for learning Mean Flows is the sum of a deterministic mean-field residual $\vr^{\texttt{sg}}_{\vtheta}(\vx_{t},t)$ and a sample-based Jacobi-velocity-product $\mathrm{sg}[\mJ]\vv^{\prime}$, where the former is the actual loss we aim to minimize given the identity in~\eqref{eq: td-meanflow-identity}. If we further expand the inner expectation, we have $\E_{\vx_{0}\mid{\vx_{t}}}[\vv^{\prime}]=0$ based on~\eqref{eq: margv-condv-relation} and the cross term vanishes, resulting in

\begin{equation}
    \mathbb{E}_{\vx_{0}\mid{\vx_{t}}}[\ell_{\text{MF}}(\vtheta)]=\mathbb{E}_{\vx_{0}\mid{\vx_{t}}}\left[\left\|\vr(\vx_{t},t)+\texttt{sg}[\mJ\vv^{\prime}]\right\|_{2}^{2}\right]=\left\|\vr_{\vtheta}(\vx_{t},t)\right\|_{2}^{2}+\texttt{sg}[\Tr(\mJ\Sigma_{\vv^\prime}\mJ^\top)],
    \label{eq: per-sample-expand}
\end{equation}

where $\Sigma_{\vv^{\prime}}=\Cov_{\vx_{0}\mid{\vx_{t}}}[\vv^{\prime}]$ is the \textit{non-zero} covariance of the conditional velocity field and $\ell_{\text{MF}}(\vtheta)$ is the per-sample loss. Replacing marginal velocity fields with conditional ones gives birth to an \textit{irreducible per-sample variance} given by $\Tr(\Sigma_{\vv^{\prime}})$, while $\mJ$ amplifies it into a gradient variance $\Var[\nabla_{\vtheta}\ell_{\text{MF}}\mid\vx_{t}]\propto\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$ that grows quadratically in the spatial Jacobian (\cref{thm: jacobi-variance}; proof in~\Cref{appx: theorem-proof}). With large intervals $(t-r)$ or a large Jacobi norm, the per-sample gradient is dominated by the noise term $\mJ\vv^\prime$, downweighting the mean-field signal from $\vr^{\mathrm{sg}}_{\vtheta}$.

To help our readers better understand the existence of such a gap, we visualize the magnitude of this fluctuation $\|\vv^{\prime}\|$ on a example 3-Gaussian mixture in~\cref{fig: cm-gap}. It is clear that such a gap is \textit{non-zero}, \textit{increases} when approaching the latent, and concentrates spatially in a \textit{mode-mixing regions near the latent} where multiple conditional paths overlap. Moreover, the stop-gradient operator creates a \textit{semi-gradient gap}: the gradients of the MeanFlow loss with and without stop-gradient differ by a mean-field term that vanishes at convergence and a variance-driven term $\E[\nabla_{\vtheta}\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})]$ which the stop-gradient blinds (\cref{thm: semi-gradient-gap}; full statement and proof in~\Cref{appx: theorem-proof}). Without stop-gradient, this term would force $\partial_{\vx}\vu_{\vtheta}\!\to\!\frac{1}{t-r}\mI_{d}$ to suppress $\mJ$; with it, $\Tr(\mJ\Sigma_{\vv^{\prime}}\mJ^{\top})$ can grow faster than $\|\vr_{\vtheta}\|_{2}^{2}$ shrinks, producing the non-decreasing loss reported empirically.

To understand the root cause for both failure modes in \cref{thm: jacobi-variance} and \cref{thm: semi-gradient-gap}, we define the \textit{curvature gap} as the difference between the average and instantaneous velocity fields, by rearranging the identity in~\eqref{eq: td-meanflow-identity}: $\Delta(\vx_{t},r,t):=\vu(\vx_{t},r,t)-\vv(\vx_{t},t)=-(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu$. The mean-field residual in~\eqref{eq: expanded-mf-loss} can therefore be rewritten as $\vr^{\texttt{sg}}_{\vtheta}=\Delta_{\vtheta}+(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu_{\vtheta}$, where we simplify our notation and denote $\Delta_{\vtheta}:=\vu_{\vtheta}-\vv$. At convergence the total derivative should exactly cancel the curvature gap, so the training signal $\vr^{\texttt{sg}}_{\vtheta}$ is a small difference of two quantities whose magnitudes scale with $\left\|\Delta\right\|$. However, when $\left\|\Delta\right\|$ is large, this cancellation becomes delicate. Specifically, the gradient signal from $\vr_{\vtheta}$ is weak relative to the noise $\mJ\vv^{\prime}$, while the stop-gradient operator  prevents the optimizer from controlling the resulting variance growth. This raises a natural question: how large is $\Delta$ in practice, and what governs its magnitude?

\subsection{Flow Acceleration}
\label{subsec: flow-acc}

The curvature gap $\Delta(\vx_{t},r,t)$ controls the severity of both Jacobian variance amplification and the semi-gradient gap. Quantifying the gap using a Taylor expansion of the velocity field along the flow yields the leading-order identity
\begin{equation}
    \Delta(\vx_{t},r,t)=-\frac{(t-r)}{2}\frac{\mathrm{D}\vv}{\mathrm{D}t}(\vx_{t},t)+\gR_{2}(\vx_{t},r,t),
    \label{eq: material-derivative-expansion}
\end{equation}
where $\frac{\mathrm{D}\vv}{\mathrm{D}t}\!\triangleq\!\partial_{t}\vv+(\nabla_{\vx}\vv)\vv$ is the material derivative (\textit{a.k.a.\ flow acceleration}) and $\gR_{2}\!=\!\gO((t-r)^{2})$ is the second-order remainder. The $C^{2}$-smoothness assumption is benign because existing work uses backbones~\cite{ronneberger2015unet,nichol2021improved} with twice-continuously-differentiable activations (e.g., GELU~\cite{hendrycks2016gaussian}); see~\cref{thm: material-derivative-expansion} and its proof in~\Cref{appx: theorem-proof}. \Cref{eq: material-derivative-expansion} clarifies prior heuristics: rectified flows~\cite{liu2022flowstraightfastlearning} straighten the marginal trajectory and drive $\frac{\mathrm{D}\vv}{\mathrm{D}t}\!\to\!\bm{0}$, eliminating $\Delta$; progressive scheduling~\cite{geng2025improvedmeanflowschallenges} avoids the problem by training with a small interval $r\!\to\!t$. The growth factor $\tfrac{t-r}{2}\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|$ is the principled criterion both heuristics target.

In addition, the relationship between the expectation gap and the flow acceleration reveals a challenge in the intermediate time step $t$, where the flow acceleration maximizes due to the intersection of conditional trajectories (as previously shown in~\cref{fig: cm-gap}). We argue that the issue can exacerbated for high-resolution, multi-modal data, where marginal velocity field changes occur rapidly.

\subsection{Variance-Aware MeanFlow}
\label{subsec: vamf}

We propose a concrete and simple training framework to, Variance-Aware MeanFlow (VaMF), that addresses all identified failure modes above. It address the Jacobian variance amplification through veclocity anchoring (\cref{subsubsec: v-anchor}) and reduces per-sample variance with adaptive weighting (\cref{subsubsec: trace-weight}).

\subsubsection{Target Network}
\label{subsubsec: v-anchor}

In~\eqref{eq: expanded-mf-loss}, the $\mJ\vv^{\prime}$ amplification arises because $\vv^{\prime}$ enters the JVP tangent, where it multiplies the spatial Jacobian $(t-r)\partial_{\vz}\vu_{\vtheta}$. We notice that, if the tangent were replaced by any \textit{deterministic estimate $\hat{\vv}$}, the fluctuation $\vv^{\prime}$ would only appear in the regression target, entering the residual linearly rather than through $\mJ$. In this case, the per-sample loss reduces to $\left\|\tilde{\vr}_{\vtheta}\right\|_{2}^{2}+\Tr(\Sigma_{\vv^{\prime}})$, where $\tilde{\vr}_{\vtheta}$ is a deterministic residual. As a result, the variance drops to the irreducible level $\Tr(\Sigma_{\vv^{\prime}})$ without any Jacobian amplification. We higlight this as a theoretical foundation missing from the proposed method in a recent study~\cite{geng2025improvedmeanflowschallenges}. The remaining question is whether we can achieve this without learning a separate velocity model.

% Therefore, to eliminate the Jacobian variance amplification identified in~\cref{thm: jacobi-variance},
We propose to replace the stochastic JVP tangent $\vv_{\text{cond}}$ with a deterministic estimate from a target network. Specifically, we use a Polyak averaging~\cite{polyak1992acceleration} (\textit{a.k.a. exponential moving average (EMA)}~\cite{tarvainen2017mean}) of the model parameters as the target network, and the new loss writes
\begin{equation}
    \mathcal{L}_{\text{MF}}^{\text{EMA}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+
    (t-r)\texttt{sg}\left[\texttt{JVP}(\vu_{\vtheta},(\vz,r,t),(\vu_{\bar{\vtheta},t},0,1))\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: ema-mf-loss}
\end{equation}
where $\vu_{\bar{\vtheta},t}=\vu(\vx_{t},t,t)$ and $\bar{\vtheta}$ is updated by $\bar{\vtheta}\gets{\mu\bar{\vtheta}+(1-\mu)\vtheta}$ with $\mu\in[0,1)$.

% \begin{remark}
%     The Polyak averaging slowly evolves $\vu_{\bar{\vtheta}}$ to the online model $\vu_{\vtheta}$. In the early stage of the training, both $\vu_{\vtheta}$ and $\vu_{\bar{\vtheta}}$ are poor estimators. Nevertheless, the anchoring provides a direct supervision that rapidly improves $\vu_{\vtheta}$, and hence indirectly improves $\vu_{\bar{\vtheta}}$.
% \end{remark}

\subsubsection{Flow-matching Anchor}
\label{subsubsec: va-obj}

The EMA tangent helps resolve the Jacobian variance amplification but introduces a transient bias when $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\neq\vv(\vx_{t},t)$. To correct this, we add a \textit{flow matching anchor} that provides direct velocity supervision at the boundary $r\to t$:
\begin{equation}
    \gL_{\text{FM}}(\vtheta)=\E_{t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}(\vx_{t},t{-}\delta,t)-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: fm-anchor}
\end{equation}
where $\delta\sim\gU[\delta_{\min},\delta_{\max}]$ with $\delta_{\max}\ll 1$. For small $\delta$, the average velocity over $[t{-}\delta,t]$ approximates the instantaneous velocity. The full objective in VaMF is given by
\begin{equation}
    \gL_{\text{VaMF}}(\vtheta)=\gL_{\text{MF}}^{\text{EMA}}(\vtheta)+\lambda\,\gL_{\text{FM}}(\vtheta),
    \label{eq: vamf-obj}
\end{equation}
where $\lambda>0$ balances the two terms. The EMA tangent makes $\tilde{\vr}^{\text{EMA}}\!:=\!V_{\vtheta}^{\text{EMA}}\!-\!\vv(\vx_{t},t)$ deterministic given $\vx_{t}$, so the gradient $\nabla_{\vtheta}\gL_{\text{MF}}^{\text{EMA}}\!=\!2\,\E[\nabla_{\vtheta}\vu_{\vtheta}\,\tilde{\vr}^{\text{EMA}}]$ no longer carries any $\mJ\vv'$ noise (\cref{prop: vamf-grad}; cf.~\cref{thm: jacobi-variance}). The transient bias decomposes as $\tilde{\vr}^{\text{EMA}}\!=\!\vr_{\vtheta}+(t\!-\!r)\,\partial_{\vz}\vu_{\vtheta}\bigl(\vu_{\bar{\vtheta}}(\vx_{t},t,t)\!-\!\vv(\vx_{t},t)\bigr)$, bounded by $(t\!-\!r)\|\partial_{\vz}\vu_{\vtheta}\|\,\|\vu_{\bar{\vtheta}}(\vx_{t},t,t)\!-\!\vv(\vx_{t},t)\|$ and shrinking as the FM anchor drives the boundary EMA velocity toward $\vv$ (\cref{prop: bias-var}; both proofs in~\Cref{appx: theorem-proof}).

\subsubsection{Adaptive Weights}
\label{subsubsec: trace-weight}

% The per-sample magnitude of the EMA loss in~\eqref{eq: ema-mf-loss} is still heterogeneous across $(\vx_{t},r,t)$: the inner expectation contains the curvature-amplified noise $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$ from~\cref{thm: jacobi-variance}.
Although the EMA tangent removes the Jacobian-amplified variance from the \emph{gradient}, samples with large $\|\mJ\|$ still produce loss values that dwarf samples in flat, low-curvature regions, biasing the empirical-risk estimator toward the noisy regime. We addresses this by \emph{adaptively weighting} each sample to equalize its contribution to the expected loss.

We derive the adaptive weight using the Gauss-Markov Theorem. In OT-CFM, we argue it is fair to treat the conditional fluctuation $\vv'$ as Gaussian noise, and derive the Best Linear Unbiased Estimator (BLUE) of the mean-field residual $\tilde{\vr}^{\text{EMA}}$ with the inverse-covariance matrix weight $(\mJ\Sigma_{\vv'}\mJ^{\!\top})^{-1}$. However, it further introduce two practical pathologies. First, the per-sample $d\!\times\!d$ matrix inversion has a cubic-cost $\gO(d^{3})$ and is intractable for high-dimensional data or latents. Meanwhile, the matrix is ill-conditioned whenever $(t{-}r)\partial_{\vz}\vu_{\vtheta}\to\mI_{d}$ and $\mJ\!\approx\!\bm{0}$, which pushes the inverse to $+\infty$.

To address these pathologies, we propose to use isotropic-noise approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$. Because $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})\!=\!\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})$, we define the per-sample \textit{trace weight}

\begin{equation}
    w(\vx_{t},r,t)\;:=\;\frac{d}{1+\sigma_{t}^{2}\,\texttt{sg}\!\left[\Tr(\mJ\mJ^{\!\top})\right]}\,,
    \label{eq: trace-weight}
\end{equation}
where $d$ is the data dimensionality.
% The additive $1$ provides the numerical stability that the BLUE inverse lacks: $w\!\to\!1$ as $\mJ\!\to\!\bm{0}$ (low curvature, no reweighting) and $w\!\to\!d/(\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top}))$ as the curvature term dominates (recovering the proportionality of the BLUE-scalar limit). The stop-gradient on $\Tr(\mJ\mJ^{\!\top})$ ensures the weight does not contribute to the parameter update, so $w$ acts purely as a per-sample reweighting.
Still, the direct evaluation of $\Tr(\mJ\mJ^{\!\top})$ requires $d$ Jacobian-vector products and is intractable for high-dimensional data. We instead use a single Rademacher probe $\vz\!\sim\!\gN(\bm{0},\mI_{d})$ and the unbiased Hutchinson's estimator~\cite{hutchinson1989stochastic}
\begin{equation}
    \widehat{\Tr(\mJ\mJ^{\!\top})}\;=\;\bigl\|\mJ\vz\bigr\|_{2}^{2}\;=\;\bigl\|(t{-}r)\,\partial_{\vz}\vu_{\vtheta}\,\vz-\vz\bigr\|_{2}^{2},
    \label{eq: hutchinson}
\end{equation}
which costs a single JVP per step. Multiple probes can be averaged when extra accuracy is needed. We find $K=1$ is sufficient for our experiment with DiT on ImageNet dataset.

In addition, the scalar $\sigma_{t}$ in~\eqref{eq: trace-weight} is the \textit{time-dependent} standard-deviation scale of the isotropic-noise approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$; the actual conditional-velocity variance depends on $\pdata$ and is generally not closed-form. Under the tractable Laplacian approximation $\vx_{0}\!\sim\!\gN(\bm{0},\sigma_{d}^{2}\mI_{d})$, we use conjugate-Gaussian inference to derive
\begin{equation}
    \Sigma_{\vv'}(\vx_{t},t)
    \;=\;\frac{\sigma_{d}^{2}}{\sigma_{d}^{2}(1-t)^{2}+t^{2}}\,\mI_{d},
    \label{eq: gauss-sigma}
\end{equation}
which is U-shaped in $t$ and peaks near $t\!=\!0.5$ where conditional paths most strongly intersect---the same regime where the Jacobian-deviation norm peaks in~\cref{thm: jacobi-variance}. Nevertheless, $\pdata$ in practice is generally non-Gaussian, we thereby treat $\sigma_{t}$ as a hyperparameter and sweep five candidates in our experiments. The trace weight $w$ has three key properties (\cref{prop: trace-weight}, proved in~\Cref{appx: theorem-proof}): \textbf{(a)} the stop-gradient on $\Tr(\mJ\mJ^{\!\top})$ keeps $w$ out of the parameter update, so the gradient inherits the structure of~\cref{prop: vamf-grad} unchanged; \textbf{(b)} the expected per-sample noise contribution $w\,\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!\le\!d$ is uniformly capped at the data dimensionality, eliminating the unbounded curvature growth from~\cref{thm: jacobi-variance}; \textbf{(c)} $w$ is strictly decreasing in $\Tr(\mJ\mJ^{\!\top})$ and, by~\cref{thm: material-derivative-expansion}, in $(t{-}r)^{2}\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|^{2}$, so high-curvature long-interval samples are automatically downweighted---subsuming inverse-variance weighting, progressive scheduling, and acceleration-aware interval sampling in a single per-sample scalar. The full training procedure is given in~\Cref{alg: va-mf}; the connection between $w$ and the acceleration-to-noise ratio $\text{ANR}\!=\!(t{-}r)\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|/\sqrt{\Tr(\Sigma_{\vv'})}$ is in~\Cref{appx: additional-results}.
