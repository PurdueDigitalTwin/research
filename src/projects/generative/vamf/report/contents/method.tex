\section{Preliminaries}
\label{sec: preliminary}

\paragraph{Flow-Matching.} Let $\gX\subset\mathbb{R}^{d}$ and $\vx$ a data sample. Flow-matching generative models~\cite{lipman2023flowmatchinggenerativemodeling,tong2024improvinggeneralizingflowbasedgenerative} learn a time-dependent diffeomorphism $\psi(\vx,t):\mathbb{R}^{d}\!\times\![0,1]\to\mathbb{R}^{d}$ from a simple distribution $\rvx_{1}\!\sim\!{p}(\rvx,1)$ to the target $\rvx_{0}\!\sim\!{p}(\rvx,0)\!=\!\pdata$ through parameterizing a velocity field $\vv(\vx,t)\!\triangleq\!\frac{\mathrm{d}}{\mathrm{d}t}\psi(\vx,t)$. Since the exact \textit{marginal velocity field} $\vv(\vx,t)$ is generally inaccessible, conditional flow matching instead learns a conditional field $\vv(\vx,t\mid\vx_{0})\!=\!\vx_{1}-\vx_{0}$ with $\vx_{t}\!=\!(1-t)\vx_{0}+t\vx_{1}$. The substitution is justified by the following lemma. See~\Cref{appx: first} for the proof.
\begin{lemma}
    Given vector fields $\vv(\vx,t\mid\vx_{0})$ generating probability paths $p(\vx,t\mid\vx_{0})$, for any $\rvx_{0}\!\sim\!p(\rvx_{0})$, the expected conditional velocity field is equal to the marginal velocity field
    \begin{equation}
        \vv(\vx,t)=\E_{\vx_{0}\sim{p(\vx_{0},t\mid\vx)}}\!\left[\vv(\vx,t\mid\vx_{0})\right].
        \label{eq: margv-condv-relation}
    \end{equation}
\end{lemma}

% MeanFlow
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\paragraph{One-step Mean Flows.} Iterative integration of $\vv(\vx,t)$ at inference is expensive and can be numerically unstable. To bypass it, Mean Flows~\cite{geng2025meanflowsonestepgenerative} learn a two-parameter average velocity field $\vu(\rvx,r,t)$ via the identity
\begin{equation}
    (t-r)\,\vu(\vx,r,t)=\int_{r}^{t}\vv(\vx,\tau)\,\mathrm{d}\tau.
    \label{eq: meanflow-identity}
\end{equation}
Differentiating with respect to the terminal time $t$ yields
\begin{equation}
    \vu(\vx,r,t)+(t-r)\tfrac{\mathrm{d}}{\mathrm{d}t}\vu(\vx,r,t)=\vv(\vx,t),\qquad
    \tfrac{\mathrm{d}}{\mathrm{d}t}\vu=\partial_{\vx}\vu\!\cdot\!\vv(\vx,t)+\partial_{t}\vu,
    \label{eq: td-meanflow-identity}
\end{equation}
where the total derivative $\frac{\mathrm{d}}{\mathrm{d}t}\vu$ is essentially a Jacobian-vector product (JVP), denoted $\texttt{JVP}(\vu,(\vx,r,t),(\vv(\vx,t),0,1))$. The original MeanFlow paper~\cite{geng2025meanflowsonestepgenerative} replaces \emph{both} occurrences of the marginal field with the conditional field $\vv_{\text{cond}}\!=\!\vx_{1}\!-\!\vx_{0}$, giving the MeanFlow loss
\begin{equation}
    \mathcal{L}_{\text{MF}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\!\left[\left\|\vu_{\vtheta}+(t-r)\,\texttt{sg}\!\left[\texttt{JVP}(\vu_{\vtheta},(\vz,r,t),(\vv_{\text{cond}},0,1))\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: mf-loss}
\end{equation}
with $r,t\!\sim\!\gU(0,1)$, $r\!\leq\!t$, $\vx_{0}\!\sim\!\pdata$, $\vx_{1}\!\sim\!\gN(0,\mI_{d})$, $\vz\!=\!(1-t)\vx_{0}+t\vx_{1}$, and the stop-gradient $\texttt{sg}[\cdot]$ avoiding double backpropagation through the JVP.

\section{Variance Reduction in Learning Mean Flows}
\label{sec: method}

The MeanFlow loss in~\eqref{eq: mf-loss} is empirically known to be \textit{non-decreasing} and to suffer from \textit{high-variance gradients}~\cite{zhang2025alphaflowunderstandingimprovingmeanflow,geng2025improvedmeanflowschallenges}. We investigate the statistical mechanism behind this pathology. Importantly, our analysis isolates the role of the conditional velocity $\vv_{\text{cond}}$ when it appears as the JVP tangent, where it acts as a Monte Carlo control variate for the inaccessible marginal velocity, with a coefficient that vanilla MeanFlow fixes to a suboptimal value. We identify the optimal coefficient and establish a connection to a family of practical fixes proposed by concurrent work.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Two roles of the conditional velocity field}
\label{subsec: two-roles}

With Reynolds decomposition, we express the conditional velocity by $\vv_{\text{cond}}\!=\!\vv(\vx_{t},t)+\vv'$ with $\E_{\vx_{0}\mid\vx_{t}}[\vv']\!=\!\bm{0}$ according to~\eqref{eq: margv-condv-relation} and $\Sigma_{\vv'}\!\triangleq\!\Cov_{\vx_{0}\mid\vx_{t}}[\vv']$. The MeanFlow loss in~\eqref{eq: mf-loss} substitutes $\vv_{\text{cond}}$ in two distinct positions, both as the \textit{regression target} $-\vv_{\text{cond}}$, and as the \textit{tangent} inside the total derivative. Carrying the decomposition through per-sample loss $\ell_{\text{MF}}(\vtheta)$ yields:
\begin{equation}
    \E_{\vx_{0}\mid\vx_{t}}\!\bigl[\ell_{\text{MF}}(\vtheta)\bigr]
    =\bigl\|\vr_{\vtheta}^{\texttt{sg}}\bigr\|^{2}_{2}
    +\texttt{sg}\!\bigl[\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})\bigr],
    \label{eq: per-sample-expand}
\end{equation}
where $\mJ\!\triangleq\!(t{-}r)\partial_{\vz}\vu_{\vtheta}\!-\!\mI_{d}\!\in\!\mathbb{R}^{d\times d}$ is a \textit{Jacobi factor} determined by the Jacobian matrix $\partial_{\vz}\vu_{\vtheta}$ and $\vr_{\vtheta}^{\texttt{sg}}\!\triangleq\!\vu_{\vtheta}\!+\!(t{-}r)\,\texttt{sg}[\partial_{\vz}\vu_{\vtheta}\!\cdot\!\vv\!+\!\partial_{t}\vu_{\vtheta}]\!-\!\vv$ is the deterministic mean-field residual. Since the residual $\vr_{\vtheta}^{\texttt{sg}}$ is the exact loss we aim to minimize given by~\eqref{eq: td-meanflow-identity}, the trace term leads to the non-decreasing loss and eliminates $\textit{iff}$ the Jacobi satisfies $\partial_{\vx}\vu_{\vtheta}=(t-r)\mI_{d}$. As a result, we identify two roles of the conditional velocity field in this decomposition:

\noindent\textbf{Unbiased Target.} As the regression target, $\vv_{\text{cond}}$ injects $\vv'$ \textit{linearly} into the residual, contributing the \emph{irreducible} per-sample noise floor $\Tr(\Sigma_{\vv'})$. This contribution is independent of $\vtheta$ and reflects the natural Monte Carlo cost of replacing the inaccessible $\vv$ by a per-sample proxy.

\noindent\textbf{Variance Amplifier.} As the JVP tangent, the same $\vv_{\text{cond}}$ injects $\vv'$ \textit{multiplicatively} through the Jacobi factor $\mJ$, contributing the \emph{amplified} term $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$. It scales unboundedly with the spectral magnitude of $\mJ$. We identify this amplification the dominant driver of variance at the gradient level:

\begin{theorem}[Jacobian Variance Amplification]
    Let $\boldsymbol{g}\!\triangleq\!\nabla_{\vtheta}\vu_{\vtheta}(\vx_{t},r,t)\!\in\!\mathbb{R}^{d\times p}$ be the parameter Jacobian of the average velocity. The conditional per-step gradient variance of $\ell_{\text{MF}}(\vtheta)$ in~\eqref{eq: mf-loss} satisfies
    \begin{equation*}
        \boxed{
            \Var\!\left[\nabla_{\vtheta}\ell_{\text{MF}}\mid\vx_{t}\right]
            \;\propto\;
            \boldsymbol{g}^{\!\top}\,\mJ\,\Sigma_{\vv'}\,\mJ^{\!\top}\,\boldsymbol{g}.
        }
    \end{equation*}
    \label[theorem]{thm: jacobi-variance}
\end{theorem}

The proof, in~\Cref{appx: theorem-proof}, uses the cyclic property of the trace and the positive semi-definiteness of the parameter-space Gram matrix $\boldsymbol{g}\boldsymbol{g}^{\!\top}$.~\Cref{thm: jacobi-variance} identifies the quadratic form $\boldsymbol{g}^{\!\top}\!\mJ\Sigma_{\vv'}\mJ^{\!\top}\boldsymbol{g}$ as the central object: it grows with $\|\mJ\|^{2}$ (large when the spatial Jacobian deviates from identity), saturates only at the irreducible noise floor when $\mJ\!\to\!\bm{0}$, and is uncontrolled by any term in the loss due to the use of the stop-gradient in the original MeanFlow.

\paragraph{The semi-gradient gap.} The stop-gradient operator freezes this amplification from the optimizer by treating $\mJ$ as constant during backpropagation. Without stop-gradient, the optimizer could in principle reduce variance by driving $\mJ\!\to\!\bm{0}$. This establishes a \textit{semi-gradient gap}, formally:

\begin{theorem}[Semi-Gradient Gap]
    The gradient of the MeanFlow loss $\gL_{\text{MF}}$ with and without the stop-gradient operator differ by
    \begin{equation*}
        \boxed{
            \underbrace{2(t{-}r)\,\E\!\left[\nabla_{\vtheta}\!\left(\partial_{\vx_{t}}\vu_{\vtheta}\!\cdot\!\vv+\partial_{t}\vu_{\vtheta}\right)\!\cdot\!\vr_{\vtheta}\right]}_{\text{mean-field gradient difference}}
            +\underbrace{\E\!\left[\nabla_{\vtheta}\Tr\!\left(\mJ\Sigma_{\vv'}\mJ^{\!\top}\right)\right]}_{\text{variance-driven gradient difference}}.
        }
    \end{equation*}
    \label[theorem]{thm: semi-gradient-gap}
\end{theorem}

At convergence, the mean-field difference vanishes ($\vr_{\vtheta}\!\to\!\bm{0}$) and the variance-driven term dominates. Since the optimizer is \textit{blind} to the variance-amplification term in the original MeanFlow, the term is silently accumulating, producing the non-decreasing-loss empirical pathology. \Cref{fig: cm-gap} visualizes the an example case. We showcase that $\|\vv'\|$ is non-zero, increases toward the latent, and concentrates spatially in mode-mixing regions, contributing to the non-decreasing loss.

\subsection{Bias-variance tradeoff and the tangent as a control variate}
\label{subsec: control-variate}

The amplification in~\Cref{thm: jacobi-variance} arises because the JVP tangent $\vv_{\text{cond}}\!=\!\vv\!+\!\vv'$ injects the zero-mean fluctuation $\vv'$. This corresponds to the canonical structure of a Monte Carlo \textit{control variate}, where $\vv_{\text{cond}}$ is a stochastic estimator of the inaccessible $\vv$ with known mean, and the coefficient on the noise term $\vv'$ can be a free parameter that determines the bias-variance trade-off. We make this explicit by introducing a tangent-mixing coefficient $\beta\!\in\![0,1]$:
\begin{equation}
    \tilde{\vv}_{\text{tang}}(\beta)\;\triangleq\;(1-\beta)\,\vv_{\text{cond}} + \beta\,\hat{\vv}\;=\;\vv + (1-\beta)\,\vv' + \beta\,\vb,
    \label{eq: tangent-mix}
\end{equation}
where $\hat{\vv}$ can be any deterministic proxy for the marginal velocity (\textit{e.g., an EMA-derived estimate}) with bias $\vb\!\triangleq\!\hat{\vv}-\vv$ that is deterministic given $\vx_{t}$. Herein, $\beta\!=\!0$ recovers vanilla MeanFlow and $\beta\!=\!1$ replaces the tangent entirely with the deterministic proxy.

\paragraph{Bias and variance.} Substituting $\tilde{\vv}_{\text{tang}}(\beta)$ for $\vv_{\text{cond}}$ in the JVP tangent and propagating through the loss, the per-sample gradient writes
\begin{equation}
    \boldsymbol{g}^{(\beta)} \;=\; 2\,\boldsymbol{g}^{\!\top}\!\Bigl[\vr_{\vtheta} \;+\; \bigl((1{-}\beta)\,\mJ - \beta\,\mI\bigr)\,\vv' \;+\; \beta\,(\mJ{+}\mI)\,\vb\Bigr],
    \label{eq: g-beta}
\end{equation}
where $\vr_{\vtheta}$ is the deterministic mean-field residual, the noise term scales the conditional fluctuation $\vv'$ by the matrix $((1{-}\beta)\,\mJ - \beta\,\mI)$, and the bias term scales the proxy error $\vb$ by $\beta\,(\mJ{+}\mI)$.

\begin{theorem}[Optimal control-variate coefficient]
    Under the scalar-isotropic approximation $\Sigma_{\vv'}\!\approx\!\sigma^{2}\mI_{d}$ and $\mJ\!\approx\!\kappa\mI_{d}$, the conditional MSE of $\boldsymbol{g}^{(\beta)}$ is $M(\beta) = \beta^{2}(\kappa{+}1)^{2}\|\vb\|^{2} + \sigma^{2}d\bigl((1{-}\beta)\kappa - \beta\bigr)^{2}$ and admits a unique minimizer
    \begin{equation*}
        \boxed{
            \beta^{\star} \;=\; \underbrace{\frac{\kappa}{\kappa+1}}_{\text{noise-cancellation}}\;\cdot\;\underbrace{\frac{\sigma^{2}d}{\sigma^{2}d + \|\vb\|^{2}}}_{\text{shrinkage}},
        }
    \end{equation*}
    with optimum value $M(\beta^{\star})\!=\!\sigma^{2}d\,\kappa^{2}\,\|\vb\|^{2}/(\sigma^{2}d+\|\vb\|^{2})$. For all $\kappa,\|\vb\|^{2}\!>\!0$, $M(\beta^{\star})$ strictly dominates both corner MSEs $M(0)\!=\!\sigma^{2}\kappa^{2}d$ (vanilla MeanFlow) and $M(1)\!=\!(\kappa{+}1)^{2}\|\vb\|^{2}+\sigma^{2}d$ (deterministic-tangent estimator with bias).
    \label[theorem]{thm: control-variate-optimum}
\end{theorem}

The proof is a direct application of bias-variance decomposition to~\eqref{eq: g-beta} and is given in~\Cref{appx: theorem-proof}. The two factors of $\beta^{\star}$ have crisp interpretations: $\kappa/(\kappa{+}1)$ is the coefficient that \emph{cancels the Jacobian-amplified noise} via destructive interference between the $\mJ\vv'$ and $\mI\vv'$ contributions in~\eqref{eq: g-beta}; $\sigma^{2}d/(\sigma^{2}d{+}\|\vb\|^{2})$ is the standard James-Stein shrinkage that pulls $\beta^{\star}$ toward the unbiased corner when the proxy bias $\|\vb\|$ is large. As the proxy bias shrinks ($\|\vb\|^{2}\!\to\!0$), $\beta^{\star}\!\to\!\kappa/(\kappa{+}1)$, and as the Jacobian magnitude grows ($\kappa\!\to\!\infty$), $\beta^{\star}\!\to\!1$.

\paragraph{Vanilla MeanFlow is generally suboptimal.} Vanilla MeanFlow uses $\beta\!=\!0$ which is optimal only when $\kappa\!=\!0$ (i.e., $\partial_{\vz}\vu_{\vtheta}\!=\!\frac{1}{t-r}\mI_{d}$) or when no deterministic proxy is available ($\|\vb\|\!=\!\infty$). Both conditions fail in practice: trained backbones produce $\partial_{\vz}\vu_{\vtheta}$ matrices that are spectrally far from $\frac{1}{t-r}\mI_{d}$, and an EMA copy of $\vu_{\vtheta}$ provides a viable deterministic proxy at near-zero compute cost.

\paragraph{Concurrent fixes through this lens.} Several remedies for MeanFlow's training pathology in concurrent work can now be located in the same one-parameter family. Each fix corresponds to a different practical realization of the optimum, distinguished by \emph{which proxy} they use for $\vv$ and \emph{which value of $\beta$} the construction implicitly selects:
\begin{itemize}
    \item \textbf{Deterministic-tangent methods} (Improved MF~\cite{geng2025improvedmeanflowschallenges}, EMA tangents) take $\beta\!\to\!1$ with a learned or EMA-derived proxy $\hat{\vv}$. The amplified noise term is eliminated; the bias term $(\mJ{+}\mI)\vb$ is controlled by training the proxy.
    \item \textbf{Reparameterization methods} (TVM~\cite{zhou2026terminalvelocitymatching}) sidestep the spatial Jacobian by reparameterizing the total derivative, effectively setting $\mJ\!=\!\bm{0}$ in our framework.
    \item \textbf{Curvature-reduction methods} (Re-MeanFlow~\cite{zhang2026overcomingcurvaturebottleneckmeanflow}) reduce $\|\mJ\|$ via reflow preprocessing, lowering $\kappa$ and the amplified-noise term.
    \item \textbf{Curriculum methods} (AlphaFlow~\cite{zhang2025alphaflowunderstandingimprovingmeanflow}) avoid high-$\kappa$ regimes during training rather than reducing the variance directly.
\end{itemize}
None of these methods explicitly identify the control-variate structure or derive $\beta^{\star}$. We do, and we use the result both as an analytical lens for these methods and as the foundation for our practical training framework.

\subsection{Variance-Aware MeanFlow}
\label{subsec: vamf}

We instantiate the variance-reduction analysis as a backbone-preserving training framework, \textit{Variance-Aware MeanFlow} (VaMF). Two design choices realize $\beta\!\to\!1$ at near-zero compute overhead.

\paragraph{EMA proxy.} We take $\hat{\vv}\!=\!\vu_{\bar{\vtheta}}(\vx_{t},t,t)$ where $\bar{\vtheta}$ is a Polyak-averaged copy of $\vtheta$~\cite{polyak1992acceleration,tarvainen2017mean}, $\bar{\vtheta}\!\gets\!\mu\bar{\vtheta}\!+\!(1{-}\mu)\vtheta$ with $\mu\!\in\![0,1)$. The resulting loss replaces the conditional tangent in~\eqref{eq: mf-loss} with the EMA proxy:
\begin{equation}
    \mathcal{L}_{\text{MF}}^{\text{EMA}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\!\left[\left\|\vu_{\vtheta}+(t{-}r)\,\texttt{sg}\!\left[\texttt{JVP}\!\bigl(\vu_{\vtheta},(\vz,r,t),(\vu_{\bar{\vtheta},t},0,1)\bigr)\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: ema-mf-loss}
\end{equation}
with $\vu_{\bar{\vtheta},t}\!\triangleq\!\vu(\vx_{t},t,t;\bar{\vtheta})$. Crucially, the regression target remains $\vv_{\text{cond}}$: the tangent and target play distinct statistical roles per~\Cref{subsec: two-roles}, and only the tangent needs to be replaced.

\paragraph{Flow-matching anchor.} The proxy bias $\vb\!=\!\vu_{\bar{\vtheta},t}\!-\!\vv$ must be controlled or it propagates through the $\beta\,(\mJ{+}\mI)\vb$ bias term. We supervise the boundary condition $\vu(\vx_{t},t,t)\!=\!\vv(\vx_{t},t)$ directly with a small flow-matching loss
\begin{equation}
    \gL_{\text{FM}}(\vtheta)=\E_{t,\vx_{0},\vx_{1}}\!\left[\left\|\vu_{\vtheta}(\vx_{t},t{-}\delta,t)-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: fm-anchor}
\end{equation}
where $\delta\!\sim\!\gU[\delta_{\min},\delta_{\max}]$ with $\delta_{\max}\!\ll\!1$. For small $\delta$, $\vu(\vx_{t},t{-}\delta,t)$ approximates the instantaneous velocity, and~\eqref{eq: margv-condv-relation} guarantees $\E[\vv_{\text{cond}}\mid\vx_{t}]\!=\!\vv(\vx_{t},t)$, so $\gL_{\text{FM}}$ drives $\vu_{\vtheta}(\vx_{t},t,t)\!\to\!\vv(\vx_{t},t)$, and via EMA, $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\!\to\!\vv$ and hence $\vb\!\to\!\bm{0}$. The full VaMF loss is
\begin{equation}
    \gL_{\text{VaMF}}(\vtheta)=\gL_{\text{MF}}^{\text{EMA}}(\vtheta)+\lambda\,\gL_{\text{FM}}(\vtheta),
    \label{eq: vamf-obj}
\end{equation}
with $\lambda\!>\!0$. The training procedure is~\Cref{alg: va-mf}.

\paragraph{Optimality.} The combination of EMA proxy plus FM anchor approximates the $\beta\!\to\!1$ limit of~\Cref{thm: control-variate-optimum} at the cost of one additional JVP per step.~\Cref{prop: vamf-grad,prop: bias-var} (proofs in~\Cref{appx: theorem-proof}) characterize the resulting gradient and bias structure:

\begin{proposition}[Gradient of VaMF Loss]
    The gradient of $\gL_{\text{MF}}^{\text{EMA}}$ takes the form
    \begin{equation*}
        \nabla_{\vtheta}\gL_{\text{MF}}^{\text{EMA}}=2\,\E\!\left[\boldsymbol{g}\,\tilde{\vr}^{\text{EMA}}\right],
    \end{equation*}
    where $\tilde{\vr}^{\text{EMA}}\!\triangleq\!V_{\vtheta}^{\text{EMA}}\!-\!\vv(\vx_{t},t)$, and no $\mJ\vv'$ noise term appears.
    \label[proposition]{prop: vamf-grad}
\end{proposition}

\begin{proposition}[Bias-Variance Tradeoff]
    The EMA mean-field residual decomposes as
    \begin{equation*}
        \tilde{\vr}^{\text{EMA}}=\vr_{\vtheta}+(t{-}r)\,\partial_{\vz}\vu_{\vtheta}\!\left(\vu_{\bar{\vtheta}}(\vx_{t},t,t)-\vv(\vx_{t},t)\right),
    \end{equation*}
    where $\vr_{\vtheta}$ is the true MeanFlow residual. The FM anchor drives $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\!\to\!\vv(\vx_{t},t)$, giving $\tilde{\vr}^{\text{EMA}}\!\to\!\bm{0}$.
    \label[proposition]{prop: bias-var}
\end{proposition}

\paragraph{Why the target stays unbiased.} \Cref{prop: vamf-grad,prop: bias-var} treat the proxy $\hat{\vv}$ as a tangent-only replacement. A natural alternative is to substitute $\hat{\vv}$ for the regression target $\vv_{\text{cond}}$ as well, yielding a ``fully self-consistent'' loss in the spirit of self-distillation. The next proposition shows that the two replacements are not interchangeable: only the tangent admits a gap-attenuated bias.

\begin{proposition}[Target--Tangent Bias Asymmetry]
    Let $\hat{\vv}=\vu_{\bar{\vtheta}}(\vx_{t},t,t)$ be a deterministic proxy with bias $\vb\!\triangleq\!\hat{\vv}-\vv(\vx_{t},t)$. Consider two ways of substituting $\hat{\vv}$ into~\eqref{eq: mf-loss}:
    \begin{enumerate}
        \item[\textnormal{(i)}] \textnormal{Tangent replacement (VaMF):} keep the target $\vv_{\text{cond}}$ but replace the JVP tangent. At the loss minimum,
        \begin{equation*}
            \vu^{\star}(\vx_{t},r,t)-\vv(\vx_{t},t)=-(t{-}r)\,\partial_{\vz}\vu^{\star}\,\vb+\gO(\|\vb\|^{2}),
        \end{equation*}
        so the stationary error is $\gO((t{-}r)\,\|\partial_{\vz}\vu\|\,\|\vb\|)$ and vanishes at $r\!\to\!t$.
        \item[\textnormal{(ii)}] \textnormal{Target replacement:} substitute $\hat{\vv}$ for $\vv_{\text{cond}}$ in the regression target. At $r\!=\!t$ the loss minimum satisfies
        \begin{equation*}
            \vu^{\star}(\vx_{t},t,t)-\vv(\vx_{t},t)=\vb,
        \end{equation*}
        i.e., the stationary error equals the proxy bias \emph{exactly}, uniformly in $t$ and independent of the gap $(t{-}r)$.
    \end{enumerate}
    \label[proposition]{prop: target-tangent-asymmetry}
\end{proposition}

The proof is given in~\Cref{appx: theorem-proof}. The asymmetry has three immediate consequences. \emph{First,} VaMF's design choice---replace only the tangent, keep $\vv_{\text{cond}}$ as target---is not arbitrary: tangent-replacement bias is multiplicatively damped by the gap $(t{-}r)$, while target-replacement bias is not. \emph{Second,} the FM anchor in~\eqref{eq: fm-anchor} suffices to control the residual VaMF bias because it pins $\vu_{\vtheta}(\vx_{t},t,t)\!\to\!\vv$ at the boundary $r\!=\!t$, where any target-replacement scheme would require an analogous anchor at \emph{every} $(r,t)$. \emph{Third,} the proposition locates a failure mode in self-distillation--style MeanFlow variants that use a deterministic proxy for both roles: the loss admits the trivial fixed point $\vu^{\star}\!\equiv\!\hat{\vv}$, which is preserved under proxy updates and only escaped by external supervision pulling $\hat{\vv}$ back toward $\vv$ at every $t$. Concurrent methods that adopt this pattern indeed report sensitivity to the proxy-update schedule and the boundary regularizer~\cite{geng2025improvedmeanflowschallenges,zhang2026overcomingcurvaturebottleneckmeanflow}.

\Cref{prop: vamf-grad} confirms that the $\mJ\vv'$ amplification of~\Cref{thm: jacobi-variance} is eliminated; \Cref{prop: bias-var} confirms that the residual EMA bias is controlled by the FM anchor, recovering the asymptotic optimum of~\Cref{thm: control-variate-optimum}; \Cref{prop: target-tangent-asymmetry} explains why the FM anchor at the boundary $r\!=\!t$ alone is sufficient. An optional curvature-budget reweighting (\Cref{appx: blue-scalar}) provides a Lagrangian implementation of $\Tr(\mJ\mJ^{\!\top})\!\leq\!C$ but is empirically marginal once the EMA tangent removes the dominant amplification, so we relegate it to the appendix.
