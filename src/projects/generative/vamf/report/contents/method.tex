\section{Preliminary}
\label{sec: preliminary}

\paragraph{Flow-Matching.} Let $\gX\subset\mathbb{R}^{d}$ be the space of random variables $\rvx\in\mathbb{R}^{d}$ and $\vx$ a data sample as the realized state. The objective of flow-based generative models is to learn a time-dependent diffeomorphic map $\psi(\vx,t):\mathbb{R}^{d}\times[0,1]\to\mathbb{R}^{d}$ from a simple distribution $\rvx_{1}\sim{p}(\vx,1)$ to the target data distribution $\rvx_{0}\sim{p}(\vx,0)={\pdata}$. Specifically, this is achieved by learning a \textit{time-dependent velocity field} $\vv(\vx,t):\mathbb{R}^{d}\times[0,1]\to\mathbb{R}^{d}$:
\begin{equation}
    \vv(\vx, t)\triangleq{\frac{\mathrm{d}}{\mathrm{d}t}\psi(\vx,t)},\quad \psi(\vx,0)=\vx_{0}.
    \label{eq: velocity-field}
\end{equation}
The map $\psi(\vx,t)$ induces a \textit{time-dependent probability density path} $p(\vx, t):\mathbb{R}^{d}\times[0,1]\to\mathbb{R}_{>0}$, with $\int_{\gX}p(\vx,t)\mathrm{d}\vx$, $\forall{t}\in[0,1]$.
% through the pushforward measure $p(\vx,t)\triangleq{\left[\psi\right]_{\#}(p(\vx,1))}$, where the pushforward operator $[\cdot]_{\#}$ is defined by the change-of-variable operation
% \begin{equation}
%     \left[\psi\right]_{\#}(p(\vx,1))=p(\psi^{-1}(\vx,t))\left|\det\left[\frac{\partial}{\partial\vx}\psi^{-1}(\vx,t)\right]\right|.
%     \label{eq: pushforward}
% \end{equation}
A sufficient and necessary condition for a vector field to generate a probability path is the \textit{continuity equation}, which is a partial differential equation (PDE) that writes
\begin{equation}
    \frac{\partial}{\partial t}p(\vx,t)+
    \text{div}\left(p(\vx,t)\vv(\vx,t)\right)=0,
\end{equation}
where $\text{div}(f)$ denotes the divergence of function $f$. However, the exact \textit{marginal velocity field} $\vv(\vx,t)$ is generally \textit{inaccessible}, especially in a high-dimensional sample space. The optimal-transport conditional flow-matching (OT-CFM)~\cite{lipman2023flowmatchinggenerativemodeling,tong2024improvinggeneralizingflowbasedgenerative} alternatively learns a conditional velocity field $\vv(\vx,t\mid{\vx_{0};\vtheta})\triangleq{\vv\left((1-t)\vx_{0}+t\vx_{1};\vtheta\right)}$
% by minimizing a loss function:
% \begin{equation}
%     \mathcal{L}_{\text{FM}}(\vtheta)=
%     \E_{t\sim\gU(0,1),\vx_{0}\sim{\pdata},\vx_{1}\sim\gN(0,\mI_{d})}
%     \left\|\vv\left(\vx,t;\vtheta\right)-(\vx_{1}-\vx_{0})\right\|_{2}^{2}.
%     \label{eq: cfm-obj}
% \end{equation}
% Herein, $t\sim\gU(0,1)$ is a scalar sample from the uniform distribution defined on the range of $[0,1]$ and $\vx_{1}\sim\gN(0,\mI_{d})$ is a white noise sample from the standard normal distribution defined in $\mathbb{R}^{d}$. Meanwhile,
There exists a relationship between the marginal and conditional velocity field given by the following lemma:

\begin{lemma}
    Given vector fields $\vv(\vx,t\mid{\vx_{0}})$ that generate probability paths $p(\vx,t\mid{\vx_{0}})$, for any data distribution $\rvx_{0}\sim p(\rvx_{0})$, the marginal velocity field is equal to the expected conditional velocity field
    \begin{equation}
        \vv(\vx,t)=\E_{\vx_{0}\sim{p(\vx_0,t\mid{\vx})}}
        \left[\vv(\vx,t\mid{\vx_{0}})\right].
        \label{eq: margv-condv-relation}
    \end{equation}
\end{lemma}

See~\Cref{appx: first} for its proof.

\paragraph{One-step Mean Flows.} The issue with diffusion~\cite{ho2020denoising,song2021scorebased} and flow-matching models~\cite{lipman2023flowmatchinggenerativemodeling} is the integration. Evaluating the velocity field multiple times is not only expensive but can also lead to numerical instability. A Mean Flow model~\cite{geng2025meanflowsonestepgenerative} instead learns a two-parameter average velocity field $\vu(\rvx,r,t)$ to enable efficient sampling. It leverages the identity between the average and marginal velocity field given by
\begin{equation}
    (t-r)\vu(\vx,r,t)=\int_{\tau=r}^{t}\vv(\vx,\tau)\mathrm{d}\tau,
    \label{eq: meanflow-identity}
\end{equation}
where $r$ and $t$ are scalars, the start and end time steps along the probability density path. Taking the total derivative with respect to the terminal time step $t$ yields
\begin{equation}
    \vu(\vx,r,t)+(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu(\vx,r,t)=\vv(\vx,t),
    \label{eq: td-meanflow-identity}
\end{equation}
where the total derivative of the average velocity field with respect to $t$ can be expressed as a Jacobi-vector Product (JVP)
\begin{equation}
    \frac{\mathrm{d}}{\mathrm{d}t}\vu(\vx,r,t)
    =\partial_{\vx}\vu\cdot \vv(\vx,t)
    +\partial_{r}\vu\cdot\frac{\mathrm{d}r}{\mathrm{d}t}
    +\partial_{t}\vu\cdot\frac{\mathrm{d}t}{\mathrm{d}t}
    =\partial_{\vx}\vu\cdot \vv(\vx,t)+\partial_{t}\vu
    \label{eq: dudt}
\end{equation}

We abuse the notation and express the right-hand side by $\texttt{JVP}(\vu,(\vx,r,t),(\vv(\vx,t),0,1))$. In the original MeanFlow~\cite{geng2025meanflowsonestepgenerative}, the authors replace both the marginal velocity field in the right-hand side of~\eqref{eq: td-meanflow-identity} and inside the JVP with the conditional velocity field, and derive the MeanFlow loss function given by
\begin{equation}
    \mathcal{L}_{\text{MF}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+
    (t-r)\texttt{sg}\left[\texttt{JVP}(\vu_{\vtheta},(\vz,r,t),(\vv_{\text{cond}},0,1))\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: mf-loss}
\end{equation}
where $r\sim\gU(0,1)$, $t\sim\gU(0,1)$, $r\leq{t}$, $\vx_{0}\sim\pdata$, $\vx_{1}\sim\gN(0,\mI_{d})$, $\vz=(1-t)\vx_{0}+t\vx_{1}$, $\vv_{\text{cond}}=\vx_{1}-\vx_{0}$, and $\texttt{sg}[\cdot]$ is the stop-gradient operator. The authors highlight that stopping gradients in JVP helps eliminate the need for ``double backpropagation'' and high-order derivatives.

\section{Variance-aware Mean Flows}
\label{sec: method}

The MeanFlow loss in~\eqref{eq: mf-loss} is empirically known to be \textit{non-decreasing} and to suffer from \textit{high-variance gradients}~\cite{zhang2025alphaflowunderstandingimprovingmeanflow,geng2025improvedmeanflowschallenges}. In this section, we aim to answer the two complementary questions:

\begin{mdframed}[%
    leftline=true,
    linecolor=BrickRed,
    linewidth=2pt,
    topline=false,
    rightline=false,
    bottomline=false,
    backgroundcolor=dust!15
]
\noindent\textbf{\textsl{RQ1} (Diagnosis): What ist the root cause for the training pathologies in Mean Flows?}

Existing works diagnose MeanFlow's instability from three apparently distinct angles, such as gradient conflict~\cite{zhang2025alphaflowunderstandingimprovingmeanflow} and the curvature bottleneck~\cite{zhang2026overcomingcurvaturebottleneckmeanflow}. Are they symptoms of a single underlying mechanism, and if so, what observable in the gradients does that mechanism affect?
\end{mdframed}

\begin{mdframed}[%
    leftline=true,
    linecolor=teal,
    linewidth=2pt,
    topline=false,
    rightline=false,
    bottomline=false,
    backgroundcolor=dust!15
]
\noindent\textbf{\textsl{RQ2} (Prescription): Variance-aware loss design.}

Given the identified mechanism, what is the simplest modification that can drive the gradient variance toward its irreducible level?
\end{mdframed}

In the following,~\cref{subsec: mf-bias-vars} answers~\textsl{RQ1} at the level of the gradients. We apply Reynolds decomposition to reveal both the bias induced by stop-gradient training and the per-step gradient variance governed by the Jacobi factor $\mJ=(t-r)\partial_{\vz}\vu_{\vtheta}-\mI_{d}$, and establish a connection between the bias and the \textit{curvature gap} $\Delta=\vu-\vv$. \Cref{subsec: flow-acc} then quantifies the gap in terms of flow acceleration with a Taylor expansion, exposing the material derivative as its physical source of the curvature gap. \Cref{subsec: vamf} answers~\textsl{RQ2} by proposing our training framework, \textit{Variance-Aware MeanFlow (VaMF)}, comprising an EMA velocity anchor, a flow-matching anchor, and a closed-form trace weight, all applied on top of the original MSE objective without any architectural change. Full proofs are deferred to~\Cref{appx: theorem-proof}.

\subsection{Bias and Variances in Learning Mean Flows}
\label{subsec: mf-bias-vars}

We begin with investigating the consequences of replacing the inaccessible marginal velocity $\vv(\vx,t)$ by the conditional field $\vv_{\text{cond}}=\vx_{1}-\vx_{0}$ in both the JVP tangent and the regression target. We first apply the Reynolds decomposition of the field and let $\vv(\vx,t\mid{\vx_{0}})=\vv(\vx,t)+\vv^{\prime}$, where $\vv^{\prime}$ is a zero-mean fluctuation from the marginal velocity induced by optimal transport between latent $\vx_{1}$ and data $\vx_{0}$. The regression loss~\eqref{eq: mf-loss} then expands to
\begin{equation}
    \begin{aligned}
        \gL_{\text{MF}}(\vtheta)
        &=\E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+(t-r)\texttt{sg}\left[\partial_{\vz}\vu_{\vtheta}\cdot\vv_{\text{cond}}+\partial_{t}\vu_{\vtheta}\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right] \\
        % &=\E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+(t-r)\texttt{sg}\left[\partial_{\vz}\vu_{\vtheta}\cdot \vv(\vx_{t},t)+\partial_{t}\vu_{\vtheta}\right]-\vv(\vx_{t},t)+\texttt{sg}[\mJ]\vv^{\prime}\right\|_{2}^{2}\right]\\
        &=\E_{r,t,\vx_{t},\vx_{1}}\left[\E_{\vx_{0}\mid{\vx_{t}}}\left[\left\|\underbrace{\vu_{\vtheta}+(t-r)\texttt{sg}\left[\partial_{\vz}\vu_{\vtheta}\cdot \vv(\vx_{t},t)+\partial_{t}\vu_{\vtheta}\right]-\vv(\vx_{t},t)}_{\text{mean-field residual }\vr^{\texttt{sg}}_{\vtheta}(\vx_{t},t)}+\texttt{sg}[\mJ]\vv^{\prime}\right\|_{2}^{2}\right]\right].
    \end{aligned}
    \label{eq: expanded-mf-loss}
\end{equation}

\begin{figure}[t]
    \centering
    \includegraphics[width=\textwidth]{assets/figures/illustration.pdf}
    \caption{Conditional-marginal velocity gap on a 3-mode mixture. \textbf{Top:} the expected per-sample fluctuation $\E_{\vx_{0}\mid\vx_{t}}\|\vv(\vx_{t},t) - \vv(\vx_{t},t\mid\vx_{0})\|$, evaluated along the conditional paths, grows monotonically with $t$ as conditional trajectories disperse, with the steepest rise where $\vx_{t}$ traverses the mode-mixing region. \textbf{Bottom:} the spatial heatmap at three timesteps confirms that the gap is heterogeneous and concentrates in the mode-mixing regions where multiple conditional paths overlap.}
    \label{fig: cm-gap}
    \vspace{-1em}
\end{figure}

Herein, we abuse the notations and let $\mJ\triangleq(t-r)\partial_{\vz}\vu_{\theta}-\mI_{d}\in\mathbb{R}^{d\times d}$ denote a Jacobi factor of $\vu_{\vtheta}$ given sample $\vx_{t}=\vx_{1}-\vx_{0}$. This decomposition reveals that the original loss for learning Mean Flows is the sum of a deterministic mean-field residual $\vr^{\texttt{sg}}_{\vtheta}(\vx_{t},t)$ and a sample-based Jacobi-velocity-product $\mathrm{sg}[\mJ]\vv^{\prime}$, where the former is the actual loss we aim to minimize given the identity in~\eqref{eq: td-meanflow-identity}. If we further expand the inner expectation, based on~\eqref{eq: margv-condv-relation}, we have $\E_{\vx_{0}\mid{\vx_{t}}}[\vv^{\prime}]=0$ and the cross term vanishes, resulting in
\begin{equation}
    \mathbb{E}_{\vx_{0}\mid{\vx_{t}}}[\ell_{\text{MF}}(\vtheta)]=\mathbb{E}_{\vx_{0}\mid{\vx_{t}}}\left[\left\|\vr(\vx_{t},t)+\texttt{sg}[\mJ\vv^{\prime}]\right\|_{2}^{2}\right]=\left\|\vr_{\vtheta}(\vx_{t},t)\right\|_{2}^{2}+\texttt{sg}[\Tr(\mJ\Sigma_{\vv^\prime}\mJ^\top)],
\end{equation}
where $\Sigma_{\vv^{\prime}}=\Cov_{\vx_{0}\mid{\vx_{t}}}[\vv^{\prime}]$ is the \textit{non-zero} covariance of the conditional velocity field and $\ell_{\text{MF}}(\vtheta)$ is the per-sample loss. Therefore, replacing marginal velocity fields wih conditional ones gives birth to an \textit{irreducible per-sample variance} of the conditional fluctuation given by the trace $\Tr(\Sigma_{\vv^{\prime}})$, where the matrix $\mJ$ converts it into a per-parameter loss contribution that grows quadratically in the spatial Jacobian. This establishes the following theorem.

\begin{theorem}[Jacobian Variance Amplification.]
    The Jacobi factor amplifies the variances of the original MeanFlow loss
    \begin{equation*}
        \boxed{\Var\left[\nabla_{\vtheta}\ell_{\text{MF}}(\vtheta)\mid{\vx_{t}}\right]\propto\Tr(\mJ\Sigma_{\vv^\prime}\mJ^{\!\top})}
    \end{equation*}
    \label[theorem]{thm: jacobi-variance}
\end{theorem}

\Cref{thm: jacobi-variance} explains the instability and high variance observed in training the MeanFlow model. With large intervals $(t-r)$ or large norm of the Jacobi term $\left\|\partial_{\vx_{t}}u_{\theta}\right\|_{2}^{2}$, the per-sample gradient is dominated by the noise term $\mJ\vv^\prime$, downweighting the mean-field signal from $\vr^{\mathrm{sg}}_{\vtheta}(\vx_{t},t)$.

To help our readers better understand the existence of such a gap, we visualize the magnitude of this fluctuation $\|\vv^{\prime}\|$ on a example 3-Gaussian mixture in~\cref{fig: cm-gap}. It is clear that such a gap is \textit{non-zero}, \textit{increases} when approaching the latent, and concentrates spatially in a \textit{mode-mixing regions near the latent} where multiple conditional paths overlap. Meanwhile, since the inner expectation includes the Jacobian amplified trace, we find that the stop-gradient operator creates a \textit{semi-gradient gap}.

\begin{theorem}[Semi-Gradient Gap]
    The gradients of MeanFlow loss with and without the stop-gradient operator differ by
    \begin{equation*}
        \boxed{
            \underbrace{2(t-r)\E_{r,t,\vx_{0},\vx_{1}}\left[\nabla_{\vtheta}\left(\partial_{\vx_{t}}\vu_{\theta}\cdot\vv(\vx_{t},t)+\partial_{t}\vu_{\theta}\right)\cdot\vr_{\theta}\right]}_{\text{mean-field gradient difference}}
            +\underbrace{\E_{r,t,\vx_{0},\vx_{1}}\left[\nabla_{\vtheta}\Tr(\mJ\Sigma_{\vv^{\prime}}\mJ^\top)\right]}_{\text{variance-driven gradient difference}}
        }
    \end{equation*}
    \label[theorem]{thm: semi-gradient-gap}
\end{theorem}

Herein, the stop-gradient eliminates the trace term from the expected gradient and $\vr_{\vtheta}\to\bm{0}$ at convergence. Therefore, the variance-driven gradient difference dominates the semi-gradient gap near convergence. Without stop-gradient, $\nabla_{\vtheta}\Tr(\mJ\Sigma_{\vv^{\prime}}\mJ^{\!\top})$ will force the system to minimize $\left\|(t-r)\partial_{\vx}\vu_{\vtheta}-\mI_{d}\right\|$, essentially pushes $\partial_{\vx}\vu_{\vtheta}\to\frac{1}{t-r}\mI_{d}$ to minimize the magnitude of $\mJ$. On the contrary, stop-gradient operator blinds variance-driven gradient, and as a result, $\Tr(\mJ\Sigma_{\vv^{\prime}}\mJ^{\top})$ can potentially increase faster than the magnitude of decrease in $\left\|\vr_{\vtheta}\right\|_{2}^{2}$, eventually leading to the non-decreasing loss in learning Mean Flows.

To understand the root cause for both failure modes in \cref{thm: jacobi-variance} and \cref{thm: semi-gradient-gap}, we define the \textit{curvature gap} as the difference between the average and instantaneous velocity fields, by rearranging the identity in~\eqref{eq: td-meanflow-identity}:

\begin{equation}
    \Delta(\vx_{t},r,t):=\vu(\vx_{t},r,t)-\vv(\vx_{t},t)=-(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu
    \label{eq: curvature-gap}
\end{equation}

The mean-field residual in~\eqref{eq: expanded-mf-loss} can therefore be rewritten as $\vr^{\texttt{sg}}_{\vtheta}=\Delta_{\vtheta}+(t-r)\frac{\mathrm{d}}{\mathrm{d}t}\vu_{\vtheta}$, where we simplify our notation and denote $\Delta_{\vtheta}:=\vu_{\vtheta}-\vv$. At convergence the total derivative should exactly cancel the curvature gap, so the training signal $\vr^{\texttt{sg}}_{\vtheta}$ is a small difference of two quantities whose magnitudes scale with $\left\|\Delta\right\|$. However, when $\left\|\Delta\right\|$ is large, this cancellation becomes delicate. Specifically, the gradient signal from $\vr_{\vtheta}$ is weak relative to the noise $\mJ\vv^{\prime}$, while the stop-gradient operator  prevents the optimizer from controlling the resulting variance growth. This raises a natural question: how large is $\Delta$ in practice, and what governs its magnitude?

\subsection{Flow Acceleration}
\label{subsec: flow-acc}

The curvature gap $\Delta(\vx_{t},r,t)$ defined in~\eqref{eq: curvature-gap} controls the severity of both Jacobian variance amplification and the semi-gradient gap. We now quantify $\Delta$ using flow acceleration, revealing its dependence on the material derivative of the velocity field.

\begin{theorem}[Flow Acceleration Expansion of the Expectation Gap]
    Let $\vv(\vx,t)\in{C^{2}}$ be twice continouously differentiable. The expectation gap admits a Taylor series at $\vx_{t}$:
    \begin{equation}
        \Delta(\vx_{t},r,t)=-\frac{(t-r)}{2}\frac{\mathrm{D}v}{\mathrm{D}t}(\vx_{t},t)
        +\gR_{2}(\vx_{t},r,t),
    \end{equation}
    where $\frac{\mathrm{D}v}{\mathrm{D}t}(\vx,t)\triangleq\partial_{t}\vv(\vx,t)+(\nabla_{\vx}\vv(\vx,t))\cdot \vv(\vx,t)$ is the material dervative (\textit{a.k.a. flow acceleration}) and $\gR_{2}(\vx,r,t)$ is the second-order remainder of the Taylor series.
    \label[theorem]{thm: material-derivative-expansion}
\end{theorem}

Herein, it is reasonable to assume $\vv(\vx,t)\in{C^{2}}$ since existing work primarily uses a U-Net~\cite{ronneberger2015unet,nichol2021improved} or transformer backbone with twice continuously differentiable activation functions (\textit{e.g., the GELU activation~\cite{hendrycks2016gaussian}}) to learn the velocity field. \Cref{thm: material-derivative-expansion} establishes a theoretical perspective that explains the points discussed in the existing works. Specifically, methods with rectified flows~\cite{liu2022flowstraightfastlearning} straighten the marginal ODE trajectory, thereby enforcing near-zero flow accelerations. By~\cref{thm: material-derivative-expansion}, this pushes $\Delta(\vx,r,t)\to{0}$ and eliminates the root cause for training instability. Meanwhile, methods that employ progressive scheduling~\cite{geng2025improvedmeanflowschallenges} bypasses the problem by initially training with a small interval $r\to{t}$. According to~\cref{thm: material-derivative-expansion}, the expectation gap grows at a rate factor $\frac{(t-r)}{2}\left\|\frac{\mathrm{D}}{\mathrm{D}t}(\vx,r,t)\right\|$ and hence it is reasonable to constrain the maximum interval during training by bounding this factor.

In addition, the relationship between the expectation gap and the flow acceleration reveals a challenge in the intermediate time step $t$, where the flow acceleration maximizes due to the intersection of conditional trajectories (as previously shown in~\cref{fig: cm-gap}). We argue that the issue can exacerbated for high-resolution, multi-modal data, where marginal velocity field changes occur rapidly.

\subsection{Variance-Aware MeanFlow}
\label{subsec: vamf}

We now instantiate these three design principles into a concrete training framework---Variance-Aware MeanFlow (VaMF)---that addresses all identified failure modes without modifying the backbone architecture.

\subsubsection{Velocity Anchoring (\ref{dp: var})}
\label{subsubsec: v-anchor}

In the above derivation, the $\mJ\vv^{\prime}$ amplification arises because $\vv^{\prime}$ enters through the JVP tangent, where it multiplies the spatial Jacobian $(t-r)\partial_{\vz}\vu_{\vtheta}$. We notice that, if the tangent were replaced by any \textit{deterministic estimate $\hat{\vv}$}, the fluctuation $\vv^{\prime}$ would only appear in the regression target, entering the residual linearly rather than through $\mJ$. In this case, the per-sample loss reduces to $\left\|\tilde{\vr}_{\vtheta}\right\|_{2}^{2}+\Tr(\Sigma_{\vv^{\prime}})$, where $\tilde{\vr}_{\vtheta}$ is a deterministic residual. As a result, the variance drops to the irreducible level $\Tr(\Sigma_{\vv^{\prime}})$ without any Jacobian amplification. We higlight this as a theoretical foundation missing from the proposed method in a recent study~\cite{geng2025improvedmeanflowschallenges}. Therefore, to eliminate the Jacobian variance amplification identified in~\cref{thm: jacobi-variance}, we replace the stochastic JVP tangent $\vv_{\text{cond}}$ with a deterministic estimate. Following iMF~\cite{geng2025improvedmeanflowschallenges}, we use a Polyak averaging~\cite{polyak1992acceleration} (\textit{a.k.a. exponential moving average (EMA)})~\cite{tarvainen2017mean} of the model parameters as the tangent. Specifically, the new loss writes
\begin{equation}
    \mathcal{L}_{\text{MF}}^{\text{EMA}}(\vtheta)=
    \E_{r,t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}+
    (t-r)\texttt{sg}\left[\texttt{JVP}(\vu_{\vtheta},(\vz,r,t),(\vu_{\bar{\vtheta},t},0,1))\right]-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: ema-mf-loss}
\end{equation}
where $\vu_{\bar{\vtheta},t}=\vu(\vx_{t},t,t)$ and $\bar{\vtheta}$ is updated by $\bar{\vtheta}\gets{\mu\bar{\vtheta}+(1-\mu)\vtheta}$ with $\mu\in[0,1)$.

\begin{remark}
    The Polyak averaging slowly evolves $\vu_{\bar{\vtheta}}$ to the online model $\vu_{\vtheta}$. In the early stage of the training, both $\vu_{\vtheta}$ and $\vu_{\bar{\vtheta}}$ are poor estimators. Still, the anchoring provides a direct supervision that rapidly improves $\vu_{\vtheta}$, and hence indirectly improves $\vu_{\bar{\vtheta}}$.
\end{remark}

\subsubsection{The VA-MeanFlow Objective (\ref{dp: bias})}
\label{subsubsec: va-obj}

The EMA tangent solves~\ref{dp: var} but introduces a transient bias when $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\neq\vv(\vx_{t},t)$, leaving~\ref{dp: bias} unaddressed. To correct this, we add a \textit{flow matching anchor} that provides direct velocity supervision near the boundary $r\to t$:
\begin{equation}
    \gL_{\text{FM}}(\vtheta)=\E_{t,\vx_{0},\vx_{1}}\left[\left\|\vu_{\vtheta}(\vx_{t},t{-}\delta,t)-\vv_{\text{cond}}\right\|_{2}^{2}\right],
    \label{eq: fm-anchor}
\end{equation}
where $\delta\sim\gU[\delta_{\min},\delta_{\max}]$ with $\delta_{\max}\ll 1$. For small $\delta$, the average velocity over $[t{-}\delta,t]$ approximates the instantaneous velocity, so~\eqref{eq: fm-anchor} is asymptotically equivalent to~\eqref{eq: cfm-obj} without a separate network head. The full \textbf{VA-MeanFlow} objective is:
\begin{equation}
    \gL_{\text{VA-MF}}(\vtheta)=\gL_{\text{MF}}^{\text{EMA}}(\vtheta)+\lambda\,\gL_{\text{FM}}(\vtheta),
    \label{eq: vamf-obj}
\end{equation}
where $\lambda>0$ balances the two terms. The following propositions characterize the gradient structure.

\begin{proposition}[Gradient of VA-MF Loss]
    The gradient of $\gL_{\text{MF}}^{\text{EMA}}$ takes the form
    $\nabla_{\vtheta}\gL_{\text{MF}}^{\text{EMA}}
    =2\,\E\!\left[\nabla_{\vtheta}\vu_{\vtheta}(\vx_{t},r,t)\cdot\tilde{\vr}^{\text{EMA}}\right]$,
    where $\tilde{\vr}^{\text{EMA}}\!:=V_{\vtheta}^{\text{EMA}}\!-\vv(\vx_{t},t)$ is the mean-field residual under the EMA tangent. The conditional velocity variance does not contribute.
    \label[proposition]{prop: vamf-grad}
\end{proposition}

\begin{proposition}[Bias-Variance Tradeoff]
    The EMA mean-field residual decomposes as
    $\tilde{\vr}^{\text{EMA}}=\vr_{\vtheta}
    +(t-r)\,\partial_{\vz}\vu_{\vtheta}\!\left(\vu_{\bar{\vtheta}}(\vx_{t},t,t)-\vv(\vx_{t},t)\right)$,
    where $\vr_{\vtheta}$ is the true MeanFlow residual from~\eqref{eq: expanded-mf-loss}. At convergence, $\vr_{\vtheta}\to\bm{0}$ and the FM anchor drives $\vu_{\bar{\vtheta}}(\vx_{t},t,t)\to\vv(\vx_{t},t)$, giving $\tilde{\vr}^{\text{EMA}}\to\bm{0}$.
    \label[proposition]{prop: bias-var}
\end{proposition}

\Cref{prop: vamf-grad} confirms that the VA-MF gradient has a first-order structure with no Jacobian-mediated variance amplification (cf.~\cref{thm: jacobi-variance}). \Cref{prop: bias-var} shows the transient bias is bounded by $(t-r)\|\partial_{\vz}\vu_{\vtheta}\|\|\vu_{\bar{\vtheta}}(\vx_{t},t,t)-\vv(\vx_{t},t)\|$ and shrinks as the EMA velocity improves.

\subsubsection{Trace Weight (\ref{dp: weight})}
\label{subsubsec: trace-weight}

With~\ref{dp: var} and~\ref{dp: bias} resolved, the per-sample magnitude of the EMA loss in~\eqref{eq: ema-mf-loss} is still heterogeneous across $(\vx_{t},r,t)$: the inner expectation contains the curvature-amplified noise $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})$ from~\cref{thm: jacobi-variance}. Even though the EMA tangent removes this term from the \emph{gradient}, samples with large $\|\mJ\|$ still produce loss values that dwarf samples in flat, low-curvature regions, biasing the empirical-risk estimator toward the noisy regime. Design principle~\ref{dp: weight} addresses this by \emph{reweighting} each sample to equalize its contribution to the expected loss.

\paragraph{BLUE matrix weight.} Treating the conditional fluctuation $\vv'$ as Gaussian noise, the Best Linear Unbiased Estimator (BLUE) of the mean-field residual $\tilde{\vr}^{\text{EMA}}$ uses the inverse-covariance matrix weight $(\mJ\Sigma_{\vv'}\mJ^{\!\top})^{-1}$. Two practical obstacles rule out the matrix form: \textbf{(i)} the per-sample $d\!\times\!d$ matrix inversion costs $\gO(d^{3})$ and is intractable for high-dimensional images or latents; \textbf{(ii)} the matrix is ill-conditioned whenever $(t{-}r)\partial_{\vz}\vu_{\vtheta}\!\approx\!\mI_{d}$, where $\mJ\!\approx\!\bm{0}$ pushes the inverse to $+\infty$.

\paragraph{Scalar surrogate.} Under the isotropic-noise approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$, both pathologies are absorbed by replacing the matrix weight with a single-scalar trace-based surrogate. Because $\Tr(\mJ\Sigma_{\vv'}\mJ^{\!\top})\!=\!\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})$, we define the per-sample \textit{trace weight}
\begin{equation}
    w(\vx_{t},r,t)\;:=\;\frac{1}{1+\sigma_{t}^{2}\,\texttt{sg}\!\left[\Tr(\mJ\mJ^{\!\top})\right]\big/d}\,,
    \label{eq: trace-weight}
\end{equation}
where the constant $d$ in the denominator of the curvature term is the per-sample dimensionality and the additive $1$ provides the numerical stability that the BLUE inverse lacks: $w\!\to\!1$ as $\mJ\!\to\!\bm{0}$ (low curvature, no reweighting) and $w\!\to\!d/(\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top}))$ as the curvature term dominates (recovering the proportionality of the BLUE-scalar limit). The stop-gradient on $\Tr(\mJ\mJ^{\!\top})$ ensures the weight does not contribute to the parameter update, so $w$ acts purely as a per-sample reweighting.

\paragraph{Hutchinson estimator.} Direct evaluation of $\Tr(\mJ\mJ^{\!\top})$ requires $d$ Jacobian-vector products and is intractable for image-scale data. We instead use a single Rademacher (or Gaussian) probe $\vz\!\sim\!\gN(\bm{0},\mI_{d})$ and the unbiased estimator~\cite{hutchinson1989stochastic}
\begin{equation}
    \widehat{\Tr(\mJ\mJ^{\!\top})}\;=\;\bigl\|\mJ\vz\bigr\|_{2}^{2}\;=\;\bigl\|(t{-}r)\,\partial_{\vz}\vu_{\vtheta}\,\vz-\vz\bigr\|_{2}^{2},
    \label{eq: hutchinson}
\end{equation}
which costs a single JVP per step. Multiple probes can be averaged when extra accuracy is needed; $K\!=\!1$ is sufficient for the experiments in this paper.

\paragraph{Schedule $\sigma_{t}$.} The scaling factor $\sigma_{t}$ in~\eqref{eq: trace-weight} absorbs the conditional-velocity variance scale and re-introduces a controllable schedule along the time axis. We consider three options: $\sigma_{t}\!=\!1$ (no schedule; the curvature term is the only modulation), $\sigma_{t}\!=\!t^{2}$ (default; matches the variance growth along linear interpolation paths), and a small learned MLP $\sigma_{t}\!=\!\texttt{NN}_{\phi}(t)$. We use $\sigma_{t}\!=\!t^{2}$ unless stated otherwise; the ablation in~\Cref{appx: additional-results} confirms it is the most robust across our toy datasets.

\paragraph{VaMF objective.} Combining the EMA tangent~\eqref{eq: ema-mf-loss}, FM anchor~\eqref{eq: fm-anchor}, and trace weight~\eqref{eq: trace-weight} yields the full \textbf{Variance-Aware MeanFlow} loss
\begin{equation}
    \gL_{\text{VaMF}}(\vtheta)\;=\;\E_{r,t,\vx_{0},\vx_{1}}\!\left[w(\vx_{t},r,t)\,\bigl\|\tilde{\vr}^{\text{EMA}}_{\vtheta}\bigr\|_{2}^{2}\right]\;+\;\lambda\,\gL_{\text{FM}}(\vtheta),
    \label{eq: vamf-tw-loss}
\end{equation}
which preserves the original MeanFlow MSE form---no additional network heads, no NLL.

\begin{proposition}[Trace-Weight Properties]
    Under the trace-weight loss~\eqref{eq: vamf-tw-loss} and the isotropic approximation $\Sigma_{\vv'}\!\approx\!\sigma_{t}^{2}\mI_{d}$:
    \textbf{(a)} The mean-parameter gradient is the trace-weighted form of~\cref{prop: vamf-grad} with $w$ as a stop-gradient scalar, so the BLUE-style normalization does not enter the parameter update.
    \textbf{(b)} The expected per-sample contribution to the loss decomposes into a deterministic signal $w\,\|\tilde{\vr}^{\text{EMA}}\|_{2}^{2}$ and a constant noise floor $w\,\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})\!=\!d\,\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top})/(d+\sigma_{t}^{2}\Tr(\mJ\mJ^{\!\top}))\!\le\!d$, eliminating the unbounded curvature growth from~\cref{thm: jacobi-variance}.
    \textbf{(c)} $w$ is monotonically decreasing in $\Tr(\mJ\mJ^{\!\top})$ and, by~\cref{thm: material-derivative-expansion}, in the squared interval length $(t{-}r)^{2}\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|^{2}$, so high-curvature long-interval samples are automatically downweighted---subsuming inverse-variance weighting, progressive scheduling, and acceleration-aware interval sampling in a single per-sample scalar.
    \label[proposition]{prop: trace-weight}
\end{proposition}

The full training procedure is given in~\Cref{alg: va-mf}; the connection between the trace weight and the acceleration-to-noise ratio $\text{ANR}\!=\!(t{-}r)\|\frac{\mathrm{D}\vv}{\mathrm{D}t}\|/\sqrt{\Tr(\Sigma_{\vv'})}$ is formalized in~\Cref{appx: additional-results}. At generation time only $\vu_{\vtheta}(\vx_{1},0,1)$ is needed; the trace-weight machinery exists purely to stabilize training.

\begin{algorithm}[t]
    \caption{Variance-Aware MeanFlow Training}
    \label{alg: va-mf}
    \begin{algorithmic}
        \Require Dataset $\gD$, EMA decay $\mu$, anchor weight $\lambda$, anchor interval $[\delta_{\min},\delta_{\max}]$, schedule $\sigma_{t}$, probes $K$
        \State Initialize $\bar{\vtheta}\gets\vtheta$
        \For{$k=1,2,\ldots$}
            \State Sample $(\vx_{0},\vx_{1})\sim\gD\times\gN(\bm{0},\mI_{d})$,\; $t\sim\gU[0,1]$,\; $r\sim\gU[0,t]$,\; $\delta\sim\gU[\delta_{\min},\delta_{\max}]$
            \State $\vx_{t}\gets(1-t)\vx_{0}+t\,\vx_{1}$,\quad $\vv_{\text{cond}}\gets\vx_{1}-\vx_{0}$
            \State $\vv_{\text{tang}}\gets\texttt{sg}[\vu_{\bar{\vtheta}}(\vx_{t},t,t)]$ \Comment{EMA tangent (D1)}
            \State $V_{\vtheta}\gets\vu_{\vtheta}(\vx_{t},r,t)+(t{-}r)\,\texttt{sg}[\texttt{JVP}(\vu_{\vtheta},(\vx_{t},r,t),(\vv_{\text{tang}},0,1))]$
            \State Sample $K$ probes $\{\vz_{i}\}_{i=1}^{K}\!\sim\!\gN(\bm{0},\mI_{d})$
            \State $\widehat{\Tr(\mJ\mJ^{\!\top})}\gets\frac{1}{K}\sum_{i=1}^{K}\bigl\|(t{-}r)\,\partial_{\vz}\vu_{\vtheta}(\vx_{t},r,t)\,\vz_{i}-\vz_{i}\bigr\|_{2}^{2}$
            \State $w\gets 1\,/\,\bigl(1+\sigma_{t}^{2}\,\texttt{sg}[\widehat{\Tr(\mJ\mJ^{\!\top})}]/d\bigr)$ \Comment{trace weight (D3)}
            \State $\ell_{\text{MF}}\gets w\cdot\|V_{\vtheta}-\vv_{\text{cond}}\|_{2}^{2}$
            \State $\ell_{\text{FM}}\gets\|\vu_{\vtheta}(\vx_{t},t{-}\delta,t)-\vv_{\text{cond}}\|_{2}^{2}$ \Comment{FM anchor (D2)}
            \State $\vtheta\gets\vtheta-\eta\,\nabla_{\vtheta}(\ell_{\text{MF}}+\lambda\,\ell_{\text{FM}})$
            \State $\bar{\vtheta}\gets\mu\bar{\vtheta}+(1-\mu)\vtheta$ \Comment{EMA update}
        \EndFor
        \State \Return $\vtheta$ \Comment{Generate: $\hat{\vx}_{0}=\vx_{1}-\vu_{\vtheta}(\vx_{1},0,1)$}
    \end{algorithmic}
\end{algorithm}
